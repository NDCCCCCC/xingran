```go
// EnqueueUpdateBatch 批量入队 update_user（D-01 导入路径专用变体）。
//
// ⚠ 与 EnqueueUpdateIfChanged 的语义差异（务必读完再调用）：
//   EnqueueUpdateIfChanged 做**四字段 diff**，只有 nickname/email/phone/deptId 实际变更才入队；
//   本变体**不做 diff** —— 导入场景没有"导入前旧值快照"（BatchSyncUsersToAD:679-683
//   同样承认这一点），"全量入队"才是正确语义：排水器闭包会重读 DB 后推最新值。
//
// 行为链（与 EnqueueDisable 同款）：
//   空切片 no-op → DedupeEntityIDs 保序去重（PG 对同一条 INSERT 内重复命中的行
//   DO UPDATE 会报 "cannot affect row a second time"）→ IsADSyncEnabled 前置门控
//   （无启用配置 → WARN 放行，不阻断导入）→ CreateInBatches(rows, pushQueueBatchSize) + 合并子句。
//
// ⚠ 不做 F-01 压制：压制只发生在 disable 打 update_user 上，update 之间不存在压制关系。
func (e *ADPushEnqueuer) EnqueueUpdateBatch(tx *gorm.DB, userIDs []string) error {
	if len(userIDs) == 0 {
		return nil
	}
	deduped := DedupeEntityIDs(userIDs)
	if len(deduped) == 0 {
		return nil
	}
	enabled, err := IsADSyncEnabled(tx)
	if err != nil {
		return err
	}
	if !enabled {
		applogger.Warnf("[ad_push_queue] 无启用的 AD 配置（sync_enabled=true AND status=enabled），跳过批量 update 入队: %d 个用户", len(deduped))
		return nil
	}
	now := time.Now().UTC() // Phase 58 F-12：全链路 UTC
	rows := make([]models.ADPushQueue, 0, len(deduped))
	for _, id := range deduped {
		rows = append(rows, models.ADPushQueue{
			EntityType: models.ADPushEntityUser,
			EntityID:   id,
			ActionType: models.ADPushActionUpdateUser,
			Status:     models.ADPushStatusPending,
		})
	}
	if err := tx.Clauses(adPushOnConflictClause(now)).CreateInBatches(&rows, pushQueueBatchSize).Error; err != nil {
		applogger.Errorf("[ad_push_queue] 批量 update 入队失败 (%d 行): %v", len(rows), err)
		return err // 原样上抛，调用方决定是否降级
	}
	applogger.Infof("[ad_push_queue] 批量 update 入队成功: %d 行", len(rows))
	return nil
}
```

```go
	return h.core.GetDB().Transaction(func(tx *gorm.DB) error {
		var userIDs []string
		if err := tx.WithContext(context.Background()).
			Model(&models.User{}).
			Where("username IN ?", usernames).
			Where("ad_dn IS NOT NULL AND ad_dn <> ''").
			Pluck("id", &userIDs).Error; err != nil {
			return err
		}
		if len(userIDs) == 0 {
			applogger.Infof("[AD-SYNC] 导入后无待同步用户（ad_dn 非空），跳过入队")
			return nil
		}
		return addomain.NewADPushEnqueuer().EnqueueUpdateBatch(tx, userIDs)
	})
}
```

## 1) F-07 入队侧：`BatchDelete` 加 Pluck 存在性过滤

在 `internal/services/system/user_service.go` 的 `BatchDelete` 事务内，把现状（`:643` 软删 + `:648` 入队）：
```go
		// 批量软删除用户
		if err := tx.Where("id IN ?", ids).Delete(&models.User{}).Error; err != nil {
			return apperrors.DatabaseError(err)
		}

		// Phase 57 D-04/QUEUE-06: 批量软删同事务入队 disable 动作行（一次成型 + 重复 id 去重）
		if err := addomain.NewADPushEnqueuer().EnqueueDisable(tx, ids); err != nil {
			return apperrors.DatabaseError(err)
		}
```
改为（Pluck 提到软删**之前**）：
```go
		// Phase 58 F-07: 入队前 Pluck 存在性 —— 只对真实存在的行入队。
		// 堵住"重复/已删 id 无限刷孤儿 disable 行"（排水器侧另有 user-not-found → dead 兜底，
		// 但入队侧掐根更省：一行查询 + "入队数 == 软删数"的口径可断言）。
		//
		// ⚠ 位置铁律（Phase 58 revision 修正）：必须在**软删语句之前**。
		//   `tx.Model(&models.User{})` 走 GORM 默认软删作用域，SQL 自动追加 `deleted_at IS NULL`；
		//   若把 Pluck 排在软删之后，本轮刚软删的行已被作用域过滤掉 → existing 恒空 →
		//   EnqueueDisable 变 no-op → QUEUE-06「入队数 == 软删数」不变量静默失效，
		//   且 `TestPhase57_UserService_BatchDelete_EnqueuesDisable`（Phase 57 既有用例）必红。
		//   放在软删之前时，默认作用域恰好表达"仍存在（未被软删）的行" = 本轮将被软删的那批行。
		// ⚠ 备选方案 `tx.Unscoped().Model(...)` **禁采用**：它会把历史上已软删、而本轮并未删除的
		//   id 也算进 existing，与"本轮实际删掉的集合"不再一一对应，破坏入队口径的可断言性。
		var existing []string
		if err := tx.Model(&models.User{}).Where("id IN ?", ids).Pluck("id", &existing).Error; err != nil {
			return apperrors.DatabaseError(err)
		}

		// 批量软删除用户
		if err := tx.Where("id IN ?", ids).Delete(&models.User{}).Error; err != nil {
			return apperrors.DatabaseError(err)
		}

		// Phase 57 D-04/QUEUE-06: 批量软删同事务入队 disable 动作行（一次成型 + 重复 id 去重）
		if err := addomain.NewADPushEnqueuer().EnqueueDisable(tx, existing); err != nil {
			return apperrors.DatabaseError(err)
		}
```

⚠ 不变量（写进注释与 SUMMARY）：**入队 disable 行数 == 本轮实际软删的用户数 == `len(existing)`**。Phase 57 的 `TestPhase57_UserService_BatchDelete_EnqueuesDisable` 就是这条不变量的既有守卫，本 plan 只能收紧（存在性过滤）不能破坏它。
⚠ `Delete`（单条）**不动**：`user_service.go:280` 的事务外 `First` 已经保证 id 存在，且 CONTEXT D-04 只锁 BatchDelete。
⚠ `existing` 为空时 `EnqueueDisable` 是 no-op（`len(userIDs)==0 → return nil`），无需额外分支。
