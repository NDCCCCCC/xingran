```go
// wrapTestInTransaction 在事务中运行测试函数，测试结束后回滚
func wrapTestInTransaction(t *testing.T, db *gorm.DB, testFunc func(*gorm.DB)) {
    tx := db.Begin()
    defer func() {
        // 无论测试成功或失败，都回滚事务
        r := recover()
        if r != nil {
            tx.Rollback()
            panic(r)
        } else {
            tx.Rollback()
        }
    }()

    testFunc(tx)
}
```
