```go
const CachePrefixPrefix = "cache"  // used as first segment in all Build*CacheKey
const ModuleDept = "dept"

// BuildDeptCacheKey builds dept cache keys with "cache:dept:<keyType>" format
// Example: BuildDeptCacheKey("tree:select") → "cache:dept:tree:select"
func BuildDeptCacheKey(keyType string, params ...string) string {
    parts := append([]string{CachePrefixPrefix, ModuleDept, keyType}, params...)
    return strings.Join(parts, ":")
}
```
