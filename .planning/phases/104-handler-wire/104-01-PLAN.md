```go
func HandleJSONBinding(c *gin.Context, obj interface{}) bool {
    if err := c.ShouldBindJSON(obj); err != nil {
        // D-104-3: pass through err.Error() — gin validator includes field name
        // Example: "user_name: cannot be blank"
        Error(c, apperrors.Wrap(err, apperrors.CodeParamError, err.Error()))
        return false
    }
    return true
}
```

**HandleServiceError** (D-104-4/D-104-5: apperrors business errors use business text; bare internal errors use operation+"失败"):
```go
func HandleServiceError(c *gin.Context, err error, operation string) bool {
    if err == nil {
        return true
    }
    // D-104-4: apperrors.AppError carries business code in .Code field
    // D-104-5: For non-apperrors, apply operation prefix; apperrors already has business message
    if !apperrors.IsAppError(err) {
        Error(c, err, operation+"失败")  // bare internal error — no err detail leaked
    } else {
        Error(c, err)  // apperrors — use its built-in business message
    }
    return false
}
```

Keep `HandleIDParam` and `HandleGetByID` unchanged — they are not part of the wire contract scope.
