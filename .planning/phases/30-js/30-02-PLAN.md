1. **xlsx lazy loading (D-06: triggered on import button click, not as a component):**
   - Create `ruoyi-react-frontend/src/components/import/ExcelImporter.tsx` as a thin component that exposes an `onImport` callback. Inside the click handler, do dynamic import:
     ```
     import { useState, type ChangeEvent } from 'react';
     import { Button, Spin } from 'antd';
     import { UploadOutlined } from '@ant-design/icons';

     export interface ExcelImporterProps {
       onParsed: (rows: Record<string, unknown>[]) => void | Promise<void>;
       buttonText?: string;
     }

     export function ExcelImporter({ onParsed, buttonText = '导入 Excel' }: ExcelImporterProps) {
       const [loading, setLoading] = useState(false);
       const handleChange = async (e: ChangeEvent<HTMLInputElement>) => {
         const file = e.target.files?.[0];
         if (!file) return;
         setLoading(true);
         try {
           const XLSX = await import('xlsx');
           const buf = await file.arrayBuffer();
           const wb = XLSX.read(buf, { type: 'array' });
           const sheetName = wb.SheetNames[0];
           const sheet = wb.Sheets[sheetName];
           const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(sheet);
           await onParsed(rows);
         } finally {
           setLoading(false);
           e.target.value = '';
         }
       };
       return (
         <>
           <input type="file" accept=".xlsx,.xls" onChange={handleChange} style={{ display: 'none' }} id="excel-importer-input" />
           <label htmlFor="excel-importer-input">
             <Button icon={<UploadOutlined />} loading={loading}>
               {buttonText}
             </Button>
           </label>
         </>
       );
     }
     ```
   - Update consumers (asset list page, workstation list, device import) to use `<ExcelImporter onParsed={...} />` instead of inline `import('xlsx')` or sync `import xlsx`.
