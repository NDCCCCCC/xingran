```sql
-- VDI 虚拟机细粒度权限迁移
-- 删除通用操作权限并添加细粒度权限

-- 1. 删除通用操作权限 vdi:vm:operate
DELETE FROM sys_menu WHERE perms = 'vdi:vm:operate';

-- 2. 为拥有 vdi:vm:operate 的角色自动添加全部5个细粒度权限
-- 先查询拥有 vdi:vm:operate 权限的角色
-- 注意：这个步骤在删除权限之前执行，以避免丢失权限

-- 3. 添加细粒度操作权限
INSERT INTO sys_menu (id, menu_name, parent_id, order_num, path, component, menu_type, visible, status, perms, icon, remark, created_at, updated_at, deleted_at) VALUES
('770e8400-e29b-41d4-a716-446655440020', '启动虚拟机', '770e8400-e29b-41d4-a716-446655440002', 10, '', '', 'F', '1', '0', 'vdi:vm:start', '#', '启动虚拟机操作', NOW(), NOW(), NULL),
('770e8400-e29b-41d4-a716-446655440021', '关机虚拟机', '770e8400-e29b-41d4-a716-446655440002', 11, '', '', 'F', '1', '0', 'vdi:vm:stop', '#', '关机虚拟机操作', NOW(), NOW(), NULL),
('770e8400-e29b-41d4-a716-446655440022', '重启虚拟机', '770e8400-e29b-41d4-a716-446655440002', 12, '', '', 'F', '1', '0', 'vdi:vm:restart', '#', '重启虚拟机操作', NOW(), NOW(), NULL),
('770e8400-e29b-41d4-a716-446655440023', '删除虚拟机', '770e8400-e29b-41d4-a716-446655440002', 13, '', '', 'F', '1', '0', 'vdi:vm:delete', '#', '删除虚拟机操作', NOW(), NOW(), NULL)
ON CONFLICT (id) DO NOTHING;

-- 注意：vdi:vm:sync 已存在（770e8400-e29b-41d4-a716-446655440011），无需新增

-- 4. 为拥有 vdi:vm:operate 的角色自动添加细粒度权限
-- 查询所有拥有 vdi:vm:operate 权限的角色
-- 然后为这些角色添加新的5个细粒度权限
-- 使用 SQL 的 CTE (Common Table Expression) 实现
WITH role_ids AS (
    SELECT DISTINCT ur.role_id
    FROM sys_user_role ur
    INNER JOIN sys_role_menu rm ON ur.role_id = rm.role_id
    INNER JOIN sys_menu m ON rm.menu_id = m.id
    WHERE m.perms = 'vdi:vm:operate'
),
old_operate_menu AS (
    SELECT id FROM sys_menu WHERE perms = 'vdi:vm:operate' LIMIT 1
),
new_permission_menus AS (
    SELECT id FROM sys_menu WHERE perms IN ('vdi:vm:start', 'vdi:vm:stop', 'vdi:vm:restart', 'vdi:vm:sync', 'vdi:vm:delete')
)
INSERT INTO sys_role_menu (role_id, menu_id)
SELECT r.role_id, n.id
FROM role_ids r
CROSS JOIN new_permission_menus n
ON CONFLICT (role_id, menu_id) DO NOTHING;

-- 5. 删除通用操作权限（在第4步之后执行，确保权限迁移完成）
DELETE FROM sys_role_menu 
WHERE menu_id IN (SELECT id FROM sys_menu WHERE perms = 'vdi:vm:operate');

DELETE FROM sys_menu WHERE perms = 'vdi:vm:operate';
```

**Key implementation details:**
- 删除 `vdi:vm:operate` 通用权限
- 添加4个新细粒度权限：`vdi:vm:start`, `vdi:vm:stop`, `vdi:vm:restart`, `vdi:vm:delete`
- 保留现有 `vdi:vm:sync` 权限（已在 129_add_vdi_menus.sql 中定义）
- 使用 CTE 查询为拥有 `vdi:vm:operate` 的角色自动添加全部5个细粒度权限
- 使用 `ON CONFLICT DO NOTHING` 确保幂等性
- UUID 模式：`770e8400-e29b-41d4-a716-446655440020` 到 `023`（顺序递增）
- parent_id 为 `770e8400-e29b-41d4-a716-446655440002`（虚拟机列表菜单）
- order_num 从 10 开始递增
