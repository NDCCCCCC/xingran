func (s *menuCacheService) GetTree(ctx context.Context) ([]models.Menu, error) {
    cacheKey := GetMenuTreeKey(false)
    var result []models.Menu
    expiration := s.GetExpiration(services.CacheConfigMenuTree, 30*time.Minute)
