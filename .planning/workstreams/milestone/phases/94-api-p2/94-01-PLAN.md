  - get(id: string) => Promise<BaseResponse<T>>（POST `${basePath}/${id}`，body {}）
  - create(data: Partial<CreatePayload<T>>) => POST `${basePath}`
  - update(id: string, data: Partial<CreatePayload<T>>) => POST `${basePath}/${id}/update`
  - delete(id: string) => POST `${basePath}/${id}/delete`
  - batch(action: string, data: Record<string, unknown>) => POST `${basePath}/batch`，body { action, ...data }
  - statistics(params: Record<string, unknown> = {}) => 解包 res.data ?? {}（POST `${basePath}/statistics`）
  - searchOptions(params: Record<string, unknown> = {}) => 解包 res.data ?? []（POST `${basePath}${dropdownPath}`）

- export function downloadFilePost(url: string, body: unknown, defaultFilename: string): Promise<void>（POST + responseType blob + 非 2xx 抛错 + 文件名提取 + 触发下载）
