```typescript
export const createWorkstationDeviceApi = (post: PostFn) => ({
  // 查询工位设备
  getByWorkstation: (workstationId: string) =>
    post(`/ops/workstation-device/${workstationId}`),
  
  // 手动添加设备
  addManual: (data: DeviceFormData & { workstation_id: string }) =>
    post('/ops/workstation-device/manual', data),
  
  // 同步 AD 设备
  syncAD: (workstationId: string) =>
    post('/ops/workstation-device/sync-ad', { workstation_id: workstationId }),
  
  // 同步资产设备
  syncAsset: (workstationId: string) =>
    post('/ops/workstation-device/sync-asset', { workstation_id: workstationId }),
  
  // 更新设备
  update: (id: string, data: Partial<DeviceFormData>) =>
    post(`/ops/workstation-device/${id}/update`, data),
  
  // 删除设备
  delete: (id: string) =>
    post(`/ops/workstation-device/${id}/delete`),
  
  // 设置主设备
  setPrimary: (id: string) =>
    post(`/ops/workstation-device/${id}/set-primary`)
});
```

```typescript
export const WorkstationDeviceTable: React.FC<WorkstationDeviceTableProps> = ({

  onDeviceChange

  const [devices, setDevices] = useState<WorkstationDevice[]>([]);
```

3. 实现数据加载：
```typescript
  const fetchDevices = async () => {
    setLoading(true);
    try {
      const result = await workstationDeviceApi.getByWorkstation(workstationId);
      setDevices(result.data || []);
    } catch (error) {
      message.error('加载设备失败');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchDevices();
  }, [workstationId]);
```

5. 实现操作处理函数：
```typescript
  const handleAdd = () => {
    setEditingDevice(null);
    form.resetFields();
    setModalVisible(true);
  };

  const handleEdit = (device: WorkstationDevice) => {
    setEditingDevice(device);
    form.setFieldsValue(device);
    setModalVisible(true);
  };

  const handleDelete = async (id: string) => {
    try {
      await workstationDeviceApi.delete(id);
      message.success('删除成功');
      fetchDevices();
      onDeviceChange?.();
    } catch (error) {
      message.error('删除失败');
    }
  };

  const handleSetPrimary = async (id: string) => {
    try {
      await workstationDeviceApi.setPrimary(id);
      message.success('设置成功');
      fetchDevices();
    } catch (error) {
      message.error('设置失败');
    }
  };

  const handleSyncAD = async () => {
    try {
      await workstationDeviceApi.syncAD(workstationId);
      message.success('同步成功');
      fetchDevices();
      onDeviceChange?.();
    } catch (error) {
      message.error('同步失败');
    }
  };

  const handleSyncAsset = async () => {
    try {
      await workstationDeviceApi.syncAsset(workstationId);
      message.success('同步成功');
      fetchDevices();
      onDeviceChange?.();
    } catch (error) {
      message.error('同步失败');
    }
  };

  const handleModalOk = async () => {
    try {
      const values = await form.validateFields();
      if (editingDevice) {
        await workstationDeviceApi.update(editingDevice.id, values);
        message.success('更新成功');
      } else {
        await workstationDeviceApi.addManual({ ...values, workstation_id: workstationId });
        message.success('添加成功');
      }
      setModalVisible(false);
      fetchDevices();
      onDeviceChange?.();
    } catch (error) {
      message.error(editingDevice ? '更新失败' : '添加失败');
    }
  };
```
