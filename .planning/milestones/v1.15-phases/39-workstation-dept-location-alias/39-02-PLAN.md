```go
func (s *locationAliasServiceImpl) validateAlias(ctx context.Context, alias *models.SysDeptLocationAlias) error {
    // 1. 防自映射
    if alias.DeptID == alias.LocationID {
        return errors.New("dept_id 与 location_id 不能相同(自映射)")
    }
    // 2. location 必须是外部机构 (is_external_org = 1)
    var location models.Department
    if err := s.db.WithContext(ctx).Where("id::text = ? AND deleted_at IS NULL", alias.LocationID).First(&location).Error; err != nil {
        return fmt.Errorf("物理位置部门不存在: %w", err)
    }
    if location.IsExternalOrg != 1 {
        return errors.New("物理位置必须是外部机构(is_external_org=1)")
    }
    // 3. dept 必须是 location 的后代
    var dept models.Department
    if err := s.db.WithContext(ctx).Where("id::text = ? AND deleted_at IS NULL", alias.DeptID).First(&dept).Error; err != nil {
        return fmt.Errorf("原组织部门不存在: %w", err)
    }
    // ancestors 字段是逗号分隔的 id 列表,包含该部门所有祖先(自底向上)
    // 判定: alias.LocationID 必须在 dept.Ancestors 字符串中(逗号包围,避免子串误匹配)
    if !strings.Contains(","+dept.Ancestors+",", ","+alias.LocationID+",") {
        return errors.New("原组织必须是物理位置的后代部门(ancestor 关系校验失败)")
    }
    return nil
}
```

**额外注意事项**:
- 用 `id::text = ?` 对齐 sys_dept.id (UUID) 与 alias.dept_id (varchar) 类型 (CLAUDE.md UUID Foreign Keys 约定)
- alias.scope 默认值在 service 层兜底: 若 req.Scope == "" 则赋值 "workstation"
- Create/Update 时如果 dept_id/location_id 变更, 必须重新跑 validateAlias
- 不引入新三方依赖
