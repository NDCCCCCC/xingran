```typescript
const subDeptTree = useMemo<DeptTreeNode[]>(() => {
  if (!watchedOrgId) return [];
  const node = findDeptNode(deptTreeData, watchedOrgId);
  const baseTree = node?.children?.length
    ? trimTitleToLastSegment(node.children)
    : [];

  // Phase 39 union 注入: alias 映射部门追加到末尾(D-01 决策 — 后端 is_alias=true 标记驱动)
  if (!aliasList || aliasList.length === 0) return baseTree;

  const aliasNodes: DeptTreeNode[] = aliasList.map((a) => {
    // 单节点走 trimTitleToLastSegment 保持与 baseTree title 收窄规则一致
    const trimmed = trimTitleToLastSegment([{ id: a.deptId, title: a.deptName, children: [] }])[0]?.title ?? a.deptName;
    return {
      id: a.deptId,
      title: `${trimmed} [映射]`,  // D-01: 原名 + [映射] 后缀
      isLeaf: true,
      value: a.deptId,
      // 保留 is_alias 自定义字段以便其他消费者识别(可选, 不影响渲染)
      ...({ is_alias: true } as Record<string, unknown>),
    };
  });

  return [...baseTree, ...aliasNodes];
}, [deptTreeData, watchedOrgId, aliasList]);
```
