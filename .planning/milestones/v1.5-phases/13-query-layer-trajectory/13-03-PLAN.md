// ImportOUIData 启动时从configs/oui-vendors.json导入OUI数据
func (s *macHistoryQueryServiceImpl) ImportOUIData(ctx context.Context) error {
	// 检查表是否已有数据
	var count int64
	if err := s.db.WithContext(ctx).Table("sys_mac_oui_vendor").Count(&count).Error; err != nil {
		return fmt.Errorf("failed to check OUI table: %w", err)
	}
	if count > 0 {
		log.Printf("OUI table already has %d records, skipping import", count)
		return nil
	}

	// 读取JSON文件
	data, err := os.ReadFile("configs/oui-vendors.json")
	if err != nil {
		return fmt.Errorf("failed to read OUI JSON: %w", err)
	}

	var vendors []MACOUIVendor
	if err := json.Unmarshal(data, &vendors); err != nil {
		return fmt.Errorf("failed to parse OUI JSON: %w", err)
	}

	// 批量插入（每批100条）
	batchSize := 100
	for i := 0; i < len(vendors); i += batchSize {
		end := i + batchSize
		if end > len(vendors) {
			end = len(vendors)
		}
		batch := vendors[i:end]
		if err := s.db.WithContext(ctx).Create(&batch).Error; err != nil {
			return fmt.Errorf("failed to insert OUI batch %d: %w", i/batchSize, err)
		}
	}

	log.Printf("Imported %d OUI vendors", len(vendors))
	return nil
}
```

// GetVendor 查询MAC地址的厂商信息（基于OUI前6位）
func (s *macHistoryQueryServiceImpl) GetVendor(ctx context.Context, macAddress string) (string, error) {
	// 规范化MAC地址并提取OUI前缀
	normalized := normalizeMACAddress(macAddress)
	if len(normalized) < 6 {
		return "Unknown Vendor", nil
	}
	oui := strings.ToUpper(normalized[:6]) // AABBCC格式

	// Redis缓存键
	cacheKey := fmt.Sprintf("mac:vendor:%s", oui)

	// 尝试从缓存获取
	var vendorName string
	found, err := s.cache.Get(ctx, cacheKey, &vendorName)
	if err != nil {
		log.Warnf("Redis Get failed for OUI %s: %v", oui, err)
		// 降级到DB查询
	}
	if found && vendorName != "" {
		return vendorName, nil
	}

	// DB查询
	var vendor MACOUIVendor
	if err := s.db.WithContext(ctx).Where("oui_prefix = ?", oui).First(&vendor).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			// 未知OUI，缓存"Unknown Vendor"避免重复查询
			_ = s.cache.Set(ctx, cacheKey, "Unknown Vendor", 24*time.Hour)
			return "Unknown Vendor", nil
		}
		return "", fmt.Errorf("failed to query OUI vendor: %w", err)
	}

	// 缓存结果（24小时）
	_ = s.cache.Set(ctx, cacheKey, vendor.VendorName, 24*time.Hour)

	return vendor.VendorName, nil
}
```

type GetVendorRequest struct {
	MAC string `json:"mac" binding:"required"`
}

// GetVendorResponse 厂商查询响应
type GetVendorResponse struct {
	VendorName string `json:"vendor_name"`
}

// GetVendor 查询MAC地址的厂商信息
func (h *MACHistoryHandler) GetVendor(c *gin.Context) {
	var req GetVendorRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Error(c, http.StatusBadRequest, "请求参数错误")
		return
	}

	vendorName, err := h.historyQueryService.GetVendor(c.Request.Context(), req.MAC)
	if err != nil {
		response.Error(c, http.StatusInternalServerError, err.Error())
		return
	}

	response.Success(c, GetVendorResponse{
		VendorName: vendorName,
	})
}

func TestGetVendor(t *testing.T) {
	service := setupTestService()

	// 准备测试数据
	oui := &models.MACOUIVendor{
		OUIPrefix:  "AABBCC",
		VendorName: "Test Vendor Inc",
	}

	tests := []struct {
		name       string
		mac        string
		wantVendor string
		wantErr    bool
	}{
		{
			name:       "已知OUI - 大写带分隔符",
			mac:        "AA:BB:CC:DD:EE:FF",
			wantVendor: "Test Vendor Inc",
			wantErr:    false,
		},
		{
			name:       "已知OUI - 小写无分隔符",
			mac:        "aabbccddeeff",
			wantVendor: "Test Vendor Inc",
			wantErr:    false,
		},
		{
			name:       "未知OUI",
			mac:        "DD:EE:FF:11:22:33",
			wantVendor: "Unknown Vendor",
			wantErr:    false,
		},
		{
			name:       "无效MAC格式 - 太短",
			mac:        "AA:BB",
			wantVendor: "Unknown Vendor",
			wantErr:    false,
		},
		{
			name:       "无效MAC格式 - 非十六进制",
			mac:        "GG:HH:II:JJ:KK:LL",
			wantVendor: "Unknown Vendor",
			wantErr:    false,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			vendor, err := service.GetVendor(context.Background(), tt.mac)
			if tt.wantErr {
				assert.Error(t, err)
			} else {
				assert.NoError(t, err)
				assert.Equal(t, tt.wantVendor, vendor)
			}
		})
	}
}
