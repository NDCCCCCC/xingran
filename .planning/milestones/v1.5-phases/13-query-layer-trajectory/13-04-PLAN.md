```typescript
import React, { useMemo } from 'react';
import ReactECharts from 'echarts-for-react';
import { EChartsOption } from 'echarts';
import { Empty, Spin } from 'antd';

interface TrajectoryNode {
  mac: string;
  device_name: string;
  port_name: string;
  vlan_id: number;
  first_seen: string; // ISO timestamp
  last_seen: string; // ISO timestamp
  duration_seconds: number;
  event_type: 'appeared' | 'disappeared' | 'moved' | 'vlan_changed';
}

interface MACTrajectoryChartProps {
  data: TrajectoryNode[];
  loading?: boolean;
  onError?: (error: Error) => void;
}

const EVENT_COLORS = {
  appeared: '#52c41a',     // 绿色
  disappeared: '#ff4d4f',   // 红色
  moved: '#faad14',        // 黄色
  vlan_changed: '#1890ff', // 蓝色
};

const MACTrajectoryChart: React.FC<MACTrajectoryChartProps> = ({
  data,
  loading = false,
  onError
}) => {
  // 处理错误
  if (!loading && (!data || data.length === 0)) {
    return <Empty description="暂无轨迹数据" />;
  }

  // ECharts配置
  const option: EChartsOption = useMemo(() => {
    if (!data || data.length === 0) return {};

    // 按设备分组数据
    const deviceGroups = data.reduce((groups, node) => {
      const device = node.device_name || 'Unknown Device';
      if (!groups[device]) {
        groups[device] = [];
      }
      groups[device].push(node);
      return groups;
    }, {} as Record<string, TrajectoryNode[]>);

    // 构建Gantt图数据
    const devices = Object.keys(deviceGroups);
    const seriesData = data.map((node, index) => ({
      name: `${node.port_name} (VLAN ${node.vlan_id})`,
      value: [
        devices.indexOf(node.device_name), // Y轴：设备索引
        new Date(node.first_seen).getTime(), // X轴：开始时间
        new Date(node.last_seen).getTime(),  // X轴：结束时间
        node.duration_seconds, // 停留时长（用于tooltip）
        node.event_type, // 事件类型（用于颜色）
        node.mac, // MAC地址（用于tooltip）
        node.device_name, // 设备名（用于tooltip）
      ],
      itemStyle: {
        color: EVENT_COLORS[node.event_type],
      },
    }));

    return {
      title: {
        text: `MAC轨迹: ${data[0]?.mac || 'N/A'}`,
        left: 'center',
      },
      tooltip: {
        trigger: 'item',
        formatter: (params: any) => {
          const data = params.data.value;
          const durationHours = (data[3] / 3600).toFixed(1);
          return `
            <div>
              <strong>MAC: ${data[4]}</strong><br/>
              设备: ${data[5]}<br/>
              端口: ${params.name}<br/>
              停留: ${durationHours}小时<br/>
              事件: ${data[6]}
            </div>
          `;
        },
      },
      grid: {
        left: '15%',
        right: '5%',
        bottom: '15%',
        top: '15%',
      },
      xAxis: {
        type: 'time',
        axisLabel: {
          formatter: '{yyyy}-{MM}-{dd} {HH}:{mm}',
        },
      },
      yAxis: {
        type: 'category',
        data: devices,
        axisLabel: {
          formatter: (value: string) => {
            // 设备名太长时截断
            return value.length > 15 ? value.substring(0, 12) + '...' : value;
          },
        },
      },
      series: [
        {
          type: 'custom',
          renderItem: (params: any, api: any) => {
            const categoryIndex = api.value(0);
            const timeStart = api.value(1);
            const timeEnd = api.value(2);

            const coordStart = api.coord([timeStart, categoryIndex]);
            const coordEnd = api.coord([timeEnd, categoryIndex]);

            const barHeight = 20;
            const barWidth = coordEnd[0] - coordStart[0];

            return {
              type: 'rect',
              shape: {
                x: coordStart[0],
                y: coordStart[1] - barHeight / 2,
                width: barWidth,
                height: barHeight,
              },
              style: {
                fill: api.visual('color'),
              },
            };
          },
          data: seriesData,
        },
      ],
      dataZoom: [
        {
          type: 'slider',
          xAxisIndex: 0,
          bottom: '5%',
        },
      ],
    };
  }, [data]);

  if (loading) {
    return (
      <div style={{ textAlign: 'center', padding: '50px' }}>
        <Spin tip="加载轨迹数据..." />
      </div>
    );
  }

  return (
    <div style={{ width: '100%', height: '600px' }}>
      <ReactECharts
        option={option}
        style={{ height: '100%', width: '100%' }}
        opts={{ renderer: 'canvas' }}
        onError={(error) => {
          console.error('ECharts error:', error);
          onError?.(error);
        }}
      />
    </div>
  );
};

export default MACTrajectoryChart;
```

import React, { useState } from 'react';
import { Card, DatePicker, Input, Button, Form, Space, Alert } from 'antd';
import { useQuery } from '@tanstack/react-query';
import dayjs from 'dayjs';
import MACTrajectoryChart from '@/components/network/MACTrajectoryChart';
import { queryMACTrajectory } from '@/lib/api/networkApi';

const { RangePicker } = DatePicker;

interface TrajectoryQueryParams {
  mac: string;
  start_time: string; // ISO timestamp
  end_time: string;   // ISO timestamp
}

const TrajectoryPage: React.FC = () => {
  const [form] = Form.useForm();
  const [queryParams, setQueryParams] = useState<TrajectoryQueryParams | null>(null);

  // 查询轨迹数据
  const {
    data: trajectoryData,
    isLoading,
    error,
    refetch,
  } = useQuery({
    queryKey: ['macTrajectory', queryParams],
    queryFn: () => queryMACTrajectory(queryParams!),
    enabled: !!queryParams, // 有参数时才查询
    staleTime: 5 * 60 * 1000, // 5分钟缓存
  });

  const handleSearch = async () => {
    try {
      const values = await form.validateFields();
      const { mac, dateRange } = values;

      // 格式化MAC地址（自动规范化）
      const normalizedMAC = normalizeMACAddress(mac);
      if (!normalizedMAC) {
        Alert.error('请输入有效的MAC地址');
        return;
      }

      setQueryParams({
        mac: normalizedMAC,
        start_time: dateRange[0].toISOString(),
        end_time: dateRange[1].toISOString(),
      });
    } catch (err) {
      console.error('Form validation failed:', err);
    }
  };

  const handleReset = () => {
    form.resetFields();
    setQueryParams(null);
  };

  // MAC地址规范化函数（复用后端逻辑）
  const normalizeMACAddress = (mac: string): string => {
    // 移除所有非十六进制字符
    const cleaned = mac.replace(/[^a-fA-F0-9]/g, '');
    if (cleaned.length !== 12) return '';

    // 格式化为AA:BB:CC:DD:EE:FF
    return cleaned.match(/.{2}/g)?.join(':').toUpperCase() || '';
  };

  return (
    <div style={{ padding: '24px' }}>
      <Card title="MAC地址轨迹查询" bordered={false}>
        <Form form={form} layout="inline">
          <Form.Item
            name="mac"
            label="MAC地址"
            rules={[
              { required: true, message: '请输入MAC地址' },
              {
                pattern: /^[A-Fa-f0-9]{2}[:-][A-Fa-f0-9]{2}[:-][A-Fa-f0-9]{2}[:-][A-Fa-f0-9]{2}[:-][A-Fa-f0-9]{2}[:-][A-Fa-f0-9]{2}$/,
                message: 'MAC地址格式不正确',
              },
            ]}
          >
            <Input
              placeholder="AA:BB:CC:DD:EE:FF"
              style={{ width: 200 }}
              onChange={(e) => {
                // 实时格式化输入
                const formatted = normalizeMACAddress(e.target.value);
                if (formatted) {
                  e.target.value = formatted;
                }
              }}
            />
          </Form.Item>

          <Form.Item
            name="dateRange"
            label="时间范围"
            initialValue={[dayjs().subtract(7, 'days'), dayjs()]}
            rules={[{ required: true, message: '请选择时间范围' }]}
          >
            <RangePicker
              showTime
              format="YYYY-MM-DD HH:mm"
              style={{ width: 400 }}
            />
          </Form.Item>

          <Form.Item>
            <Space>
              <Button type="primary" onClick={handleSearch}>
                查询
              </Button>
              <Button onClick={handleReset}>
                重置
              </Button>
            </Space>
          </Form.Item>
        </Form>
      </Card>

      {error && (
        <Alert
          message="查询失败"
          description={error.message}
          type="error"
          showIcon
          style={{ marginTop: '16px' }}
        />
      )}

      <div style={{ marginTop: '16px' }}>
        <MACTrajectoryChart
          data={trajectoryData || []}
          loading={isLoading}
          onError={(err) => {
            console.error('Trajectory chart error:', err);
          }}
        />
      </div>
    </div>
  );
};

export default TrajectoryPage;
```

import { post } from './api';

interface TrajectoryQueryParams {
  mac: string;
  start_time: string; // ISO timestamp
  end_time: string;   // ISO timestamp
}

interface TrajectoryNode {
  mac: string;
  device_name: string;
  port_name: string;
  vlan_id: number;
  first_seen: string;
  last_seen: string;
  duration_seconds: number;
  event_type: 'appeared' | 'disappeared' | 'moved' | 'vlan_changed';
}

interface TrajectoryResponse {
  trajectory: TrajectoryNode[];
}

/**
 * 查询MAC地址轨迹
 * @param params 查询参数（MAC地址 + 时间范围）
 * @returns 轨迹节点数组
 */
export const queryMACTrajectory = async (
  params: TrajectoryQueryParams
): Promise<TrajectoryNode[]> => {
  const result = await post<TrajectoryResponse>(
    '/network/history/trajectory',
    params
  );
  return result.data.trajectory;
};
```
