```typescript
import type {
  SyncStatus,
  SyncLog,
  SyncLogListResponse,
  SyncStatistics,
  SyncResult
} from '@/types/ad-group';

// ===== AD组同步监控 API =====

/**
 * 查询同步状态
 */
export const getSyncStatus = async (configId?: string): Promise<SyncStatus> => {
  const result = await get<SyncStatus>('/ad/groups/sync/status', {
    params: configId ? { configId } : undefined
  });
  return result.data;
};

/**
 * 手动触发全量同步
 */
export const triggerFullSync = async (configId: string): Promise<SyncResult> => {
  const result = await post<SyncResult>('/ad/groups/sync', { configId });
  return result.data;
};

/**
 * 查询同步日志列表
 */
export const getSyncLogs = async (params: {
  current?: number;
  pageSize?: number;
  configId?: string;
  syncType?: string;
  startDate?: string;
  endDate?: string;
}): Promise<SyncLogListResponse> => {
  const result = await post<SyncLogListResponse>('/ad/groups/sync/logs', params);
  return result.data;
};

/**
 * 获取同步统计
 */
export const getSyncStatistics = async (configId?: string): Promise<SyncStatistics> => {
  const result = await get<SyncStatistics>('/ad/groups/sync/statistics', {
    params: configId ? { configId } : undefined
  });
  return result.data;
};

/**
 * 确保排他性成员关系
 */
export const ensureExclusiveMembership = async (configId: string): Promise<SyncResult> => {
  const result = await post<SyncResult>('/ad/groups/exclusive', null, {
    params: { configId }
  });
  return result.data;
};
```

**API设计要点**:
1. 状态查询API
2. 手动触发API
3. 历史日志API
4. 统计数据API
5. 与后端API一一对应

```typescript
import React, { useState, useEffect, useCallback, useRef } from 'react';
import {
  Card,
  Row,
  Col,
  Statistic,
  Button,
  Table,
  Tag,
  Progress,
  Alert,
  Select,
  Space,
  Tooltip,
  message,
  Timeline,
  Descriptions
} from 'antd';
import {
  SyncOutlined,
  ReloadOutlined,
  PlayCircleOutlined,
  CheckCircleOutlined,
  CloseCircleOutlined,
  ClockCircleOutlined,
  HistoryOutlined
} from '@ant-design/icons';
import type { ColumnsType } from 'antd/es/table';
import {
  getSyncStatus,
  triggerFullSync,
  getSyncLogs,
  getSyncStatistics,
  ensureExclusiveMembership
} from '@/lib/api';
import type {
  SyncStatus,
  SyncLog,
  SyncStatistics
} from '@/types/ad-group';

const { Option } = Select;

const SyncMonitor: React.FC = () => {
  // 状态管理
  const [syncStatus, setSyncStatus] = useState<SyncStatus | null>(null);
  const [syncStatistics, setSyncStatistics] = useState<SyncStatistics | null>(null);
  const [syncLogs, setSyncLogs] = useState<SyncLog[]>([]);
  const [totalLogs, setTotalLogs] = useState(0);
  const [loading, setLoading] = useState(false);
  const [syncing, setSyncing] = useState(false);
  const [currentConfig, setCurrentConfig] = useState<string>('');
  const [logCurrent, setLogCurrent] = useState(1);
  const [logPageSize, setLogPageSize] = useState(10);

  // 自动刷新定时器
  const intervalRef = useRef<NodeJS.Timeout | null>(null);

  // 加载同步状态
  const loadSyncStatus = useCallback(async () => {
    try {
      const status = await getSyncStatus(currentConfig || undefined);
      setSyncStatus(status);
    } catch (error) {
      console.error('加载同步状态失败:', error);
    }
  }, [currentConfig]);

  // 加载统计信息
  const loadStatistics = useCallback(async () => {
    try {
      const stats = await getSyncStatistics(currentConfig || undefined);
      setSyncStatistics(stats);
    } catch (error) {
      console.error('加载统计信息失败:', error);
    }
  }, [currentConfig]);

  // 加载同步日志
  const loadSyncLogs = useCallback(async () => {
    setLoading(true);
    try {
      const response = await getSyncLogs({
        current: logCurrent,
        pageSize: logPageSize,
        configId: currentConfig || undefined
      });
      setSyncLogs(response.list);
      setTotalLogs(response.total);
    } catch (error) {
      message.error('加载同步日志失败');
    } finally {
      setLoading(false);
    }
  }, [logCurrent, logPageSize, currentConfig]);

  // 手动刷新
  const handleRefresh = () => {
    loadSyncStatus();
    loadStatistics();
    loadSyncLogs();
  };

  // 触发全量同步
  const handleTriggerSync = async () => {
    if (!currentConfig) {
      message.warning('请先选择AD配置');
      return;
    }

    setSyncing(true);
    try {
      const result = await triggerFullSync(currentConfig);
      if (result.success) {
        message.success(`同步成功: ${result.successCount}/${result.totalProcessed}`);
      } else {
        message.warning(`同步完成: ${result.successCount}成功, ${result.failedCount}失败`);
      }
      handleRefresh();
    } catch (error: any) {
      message.error(error.message || '同步失败');
    } finally {
      setSyncing(false);
    }
  };

  // 确保排他性成员关系
  const handleEnsureExclusive = async () => {
    if (!currentConfig) {
      message.warning('请先选择AD配置');
      return;
    }

    try {
      const result = await ensureExclusiveMembership(currentConfig);
      message.success(`清理完成: ${result.message}`);
      handleRefresh();
    } catch (error: any) {
      message.error(error.message || '清理失败');
    }
  };

  // 初始化加载和自动刷新
  useEffect(() => {
    handleRefresh();

    // 设置自动刷新（每30秒）
    intervalRef.current = setInterval(() => {
      loadSyncStatus();
    }, 30000);

    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, [loadSyncStatus]);

  // 日志表格列定义
  const logColumns: ColumnsType<SyncLog> = [
    {
      title: '同步时间',
      dataIndex: 'startTime',
      key: 'startTime',
      width: 180,
      render: (time: string) => new Date(time).toLocaleString('zh-CN')
    },
    {
      title: '类型',
      dataIndex: 'syncType',
      key: 'syncType',
      width: 100,
      render: (type: string) => (
        <Tag color={type === 'full' ? 'blue' : 'green'}>
          {type === 'full' ? '全量' : '增量'}
        </Tag>
      )
    },
    {
      title: '组数/成员数',
      key: 'counts',
      width: 150,
      render: (_, record) => `${record.totalGroups}/${record.totalMembers}`
    },
    {
      title: '成功/失败',
      key: 'result',
      width: 150,
      render: (_, record) => (
        <Space>
          <Tag color="success">{record.successCount}</Tag>
          <Tag color="error">{record.failedCount}</Tag>
        </Space>
      )
    },
    {
      title: '耗时',
      dataIndex: 'duration',
      key: 'duration',
      width: 100,
      render: (duration?: number) => duration ? `${duration}ms` : '-'
    },
    {
      title: '状态',
      key: 'status',
      width: 100,
      render: (_, record) => {
        if (record.errorMessage) {
          return <Tag icon={<CloseCircleOutlined />} color="error">失败</Tag>;
        }
        return <Tag icon={<CheckCircleOutlined />} color="success">成功</Tag>;
      }
    },
    {
      title: '错误信息',
      dataIndex: 'errorMessage',
      key: 'errorMessage',
      ellipsis: true,
      render: (msg?: string) => msg ? <Alert message={msg} type="error" showIcon /> : '-'
    }
  ];

  return (
    <div style={{ padding: 24 }}>
      {/* 顶部操作栏 */}
      <Card style={{ marginBottom: 24 }}>
        <Row gutter={16} align="middle">
          <Col span={8}>
            <Space>
              <span>AD配置:</span>
              <Select
                style={{ width: 200 }}
                placeholder="选择AD配置"
                value={currentConfig}
                onChange={setCurrentConfig}
              >
                {/* TODO: 从API加载AD配置列表 */}
                <Option value="config-1">AD配置1</Option>
              </Select>
            </Space>
          </Col>
          <Col span={16} style={{ textAlign: 'right' }}>
            <Space>
              <Button
                icon={<ReloadOutlined />}
                onClick={handleRefresh}
              >
                刷新
              </Button>
              <Button
                type="primary"
                icon={<PlayCircleOutlined />}
                onClick={handleTriggerSync}
                loading={syncing}
                disabled={!syncStatus?.enabled}
              >
                手动同步
              </Button>
              <Button
                danger
                icon={<SyncOutlined />}
                onClick={handleEnsureExclusive}
              >
                清理多组成员
              </Button>
            </Space>
          </Col>
        </Row>
      </Card>

      {/* 同步状态概览 */}
      <Row gutter={16} style={{ marginBottom: 24 }}>
        <Col span={6}>
          <Card>
            <Statistic
              title="同步状态"
              value={syncStatus?.enabled ? '已启用' : '已禁用'}
              valueStyle={{ color: syncStatus?.enabled ? '#3f8600' : '#cf1322' }}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic
              title="总映射数"
              value={syncStatus?.totalMappings || 0}
              prefix={<SyncOutlined />}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic
              title="活动映射"
              value={syncStatus?.activeMappings || 0}
              valueStyle={{ color: '#3f8600' }}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic
              title="上次同步"
              value={syncStatus?.lastSyncTime ? new Date(syncStatus.lastSyncTime).toLocaleDateString('zh-CN') : '-'}
              prefix={<ClockCircleOutlined />}
            />
          </Card>
        </Col>
      </Row>

      {/* 统计信息 */}
      {syncStatistics && (
        <Row gutter={16} style={{ marginBottom: 24 }}>
          <Col span={6}>
            <Card>
              <Statistic
                title="总同步次数"
                value={syncStatistics.totalSyncs}
              />
            </Card>
          </Col>
          <Col span={6}>
            <Card>
              <Statistic
                title="成功率"
                value={syncStatistics.successRate}
                suffix="%"
                valueStyle={{ color: syncStatistics.successRate > 90 ? '#3f8600' : '#cf1322' }}
              />
            </Card>
          </Col>
          <Col span={6}>
            <Card>
              <Statistic
                title="平均耗时"
                value={syncStatistics.avgDuration}
                suffix="ms"
              />
            </Card>
          </Col>
          <Col span={6}>
            <Card>
              <Statistic
                title="处理成员总数"
                value={syncStatistics.totalMembersProcessed}
              />
            </Card>
          </Col>
        </Row>
      )}

      {/* 同步日志 */}
      <Card
        title={
          <Space>
            <HistoryOutlined />
            <span>同步日志</span>
          </Space>
        }
        extra={
          <Tag color={syncStatus?.enabled ? 'success' : 'default'}>
            {syncStatus?.enabled ? '自动同步运行中' : '自动同步已停止'}
          </Tag>
        }
      >
        <Table
          columns={logColumns}
          dataSource={syncLogs}
          rowKey="id"
          loading={loading}
          pagination={{
            current: logCurrent,
            pageSize: logPageSize,
            total: totalLogs,
            showSizeChanger: true,
            showTotal: (t) => `共 ${t} 条`,
            onChange: (page, size) => {
              setLogCurrent(page);
              setLogPageSize(size || 10);
            }
          }}
        />
      </Card>
    </div>
  );
};

export default SyncMonitor;
```
