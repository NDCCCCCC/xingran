```go
package addomain

import (
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"
	"github.com/xingran-next/xingran-go-backend/internal/services/addomain"
	"github.com/xingran-next/xingran-go-backend/pkg/response"
)

// GroupSyncHandler AD组同步API处理器
type GroupSyncHandler struct {
	adService *addomain.ADDomainService
}

// NewGroupSyncHandler 创建处理器
func NewGroupSyncHandler(adService *addomain.ADDomainService) *GroupSyncHandler {
	return &GroupSyncHandler{
		adService: adService,
	}
}

// SyncAllMembersRequest 同步所有成员请求
type SyncAllMembersRequest struct {
	ConfigID string `json:"configId" binding:"required"`
}

// CreateMappingRequest 创建映射请求
type CreateMappingRequest struct {
	DeptID      string `json:"deptId" binding:"required,uuid"`
	ADGroupID   string `json:"adGroupId" binding:"required"`
	ADConfigID  string `json:"adConfigId" binding:"required,uuid"`
	MappingType string `json:"mappingType" binding:"omitempty,oneof=manual auto"`
	SyncEnabled bool   `json:"syncEnabled"`
}

// UpdateMappingRequest 更新映射请求
type UpdateMappingRequest struct {
	MappingStatus *string `json:"mappingStatus" binding:"omitempty,oneof=active inactive"`
	SyncEnabled   *bool   `json:"syncEnabled"`
}

// AutoMapRequest 自动映射请求
type AutoMapRequest struct {
	ConfigID string   `json:"configId" binding:"required,uuid"`
	DeptIDs  []string `json:"deptIds" binding:"omitempty,dive,uuid"`
}

// SyncAllMembers 手动触发同步所有成员
// @Summary 手动触发AD组成员同步
// @Description 同步所有有映射的部门的成员到对应的AD组
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param request body SyncAllMembersRequest true "同步请求"
// @Success 200 {object} response.Response "同步成功"
// @Router /api/v1/ad/groups/sync [post]
func (h *GroupSyncHandler) SyncAllMembers(c *gin.Context) {
	var req SyncAllMembersRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Error(c, http.StatusBadRequest, "请求参数错误: "+err.Error())
		return
	}

	result, err := h.adService.MemberSync.SyncAllMembers(c.Request.Context(), req.ConfigID)
	if err != nil {
		response.Error(c, http.StatusInternalServerError, "同步失败: "+err.Error())
		return
	}

	response.Success(c, result)
}

// GetSyncStatus 查询同步状态
// @Summary 查询AD组同步状态
// @Description 获取当前同步任务的状态和统计信息
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param configId query string false "AD配置ID"
// @Success 200 {object} response.Response "查询成功"
// @Router /api/v1/ad/groups/sync/status [get]
func (h *GroupSyncHandler) GetSyncStatus(c *gin.Context) {
	configID := c.Query("configId")

	// 获取映射统计
	mappings, err := h.adService.DeptGroupMapping.ListMappings(c.Request.Context(), &addomain.ListMappingsRequest{
		ADConfigID:    &configID,
		MappingStatus: (*addomain.MappingStatus)(&[]string{"active"}[0]),
		Current:       1,
		PageSize:      1,
	})
	if err != nil {
		response.Error(c, http.StatusInternalServerError, "查询状态失败: "+err.Error())
		return
	}

	status := map[string]interface{}{
		"enabled":        h.adService.GroupConfig.IsGroupSyncEnabled(c.Request.Context()),
		"totalMappings":  mappings.Total,
		"activeMappings": mappings.Total,
	}

	response.Success(c, status)
}

// ListMappings 查询部门-组映射列表
// @Summary 查询部门-组映射列表
// @Description 分页查询部门与AD组的映射关系
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param current query int false "页码" default(1)
// @Param pageSize query int false "每页数量" default(10)
// @Param configId query string false "AD配置ID"
// @Param deptId query string false "部门ID"
// @Param mappingType query string false "映射类型"
// @Success 200 {object} response.Response "查询成功"
// @Router /api/v1/ad/groups/mappings [get]
func (h *GroupSyncHandler) ListMappings(c *gin.Context) {
	current, _ := strconv.Atoi(c.DefaultQuery("current", "1"))
	pageSize, _ := strconv.Atoi(c.DefaultQuery("pageSize", "10"))

	var configID, deptID, mappingType *string
	if ci := c.Query("configId"); ci != "" {
		configID = &ci
	}
	if di := c.Query("deptId"); di != "" {
		deptID = &di
	}
	if mt := c.Query("mappingType"); mt != "" {
		mappingType = &mt
	}

	req := &addomain.ListMappingsRequest{
		ADConfigID:    configID,
		DeptID:        deptID,
		MappingType:   (*addomain.MappingType)(mappingType),
		MappingStatus: (*addomain.MappingStatus)(&[]string{"active"}[0]),
		Current:       current,
		PageSize:      pageSize,
	}

	result, err := h.adService.DeptGroupMapping.ListMappings(c.Request.Context(), req)
	if err != nil {
		response.Error(c, http.StatusInternalServerError, "查询失败: "+err.Error())
		return
	}

	response.Success(c, result)
}

// CreateMapping 创建部门-组映射
// @Summary 创建部门-组映射
// @Description 手动创建部门与AD组的映射关系
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param request body CreateMappingRequest true "创建映射请求"
// @Success 200 {object} response.Response "创建成功"
// @Router /api/v1/ad/groups/mappings [post]
func (h *GroupSyncHandler) CreateMapping(c *gin.Context) {
	var req CreateMappingRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Error(c, http.StatusBadRequest, "请求参数错误: "+err.Error())
		return
	}

	if req.MappingType == "" {
		req.MappingType = "manual"
	}

	createReq := &addomain.CreateMappingRequest{
		DeptID:      req.DeptID,
		ADGroupID:   req.ADGroupID,
		ADConfigID:  req.ADConfigID,
		MappingType: addomain.MappingType(req.MappingType),
		SyncEnabled: req.SyncEnabled,
		CreatedBy:   "admin", // TODO: 从JWT获取用户ID
	}

	mapping, err := h.adService.DeptGroupMapping.CreateMapping(c.Request.Context(), createReq)
	if err != nil {
		response.Error(c, http.StatusInternalServerError, "创建失败: "+err.Error())
		return
	}

	response.Success(c, mapping)
}

// GetMapping 获取单个映射详情
// @Summary 获取映射详情
// @Description 根据ID查询部门-组映射详情
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param id path string true "映射ID"
// @Success 200 {object} response.Response "查询成功"
// @Router /api/v1/ad/groups/mappings/{id} [get]
func (h *GroupSyncHandler) GetMapping(c *gin.Context) {
	id := c.Param("id")

	mapping, err := h.adService.DeptGroupMapping.GetMapping(c.Request.Context(), id)
	if err != nil {
		response.Error(c, http.StatusNotFound, "映射不存在")
		return
	}

	response.Success(c, mapping)
}

// UpdateMapping 更新部门-组映射
// @Summary 更新部门-组映射
// @Description 更新映射的状态和同步开关
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param id path string true "映射ID"
// @Param request body UpdateMappingRequest true "更新请求"
// @Success 200 {object} response.Response "更新成功"
// @Router /api/v1/ad/groups/mappings/{id} [put]
func (h *GroupSyncHandler) UpdateMapping(c *gin.Context) {
	id := c.Param("id")
	var req UpdateMappingRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Error(c, http.StatusBadRequest, "请求参数错误: "+err.Error())
		return
	}

	updateReq := &addomain.UpdateMappingRequest{
		MappingStatus: (*addomain.MappingStatus)(req.MappingStatus),
		SyncEnabled:   req.SyncEnabled,
		UpdatedBy:     "admin", // TODO: 从JWT获取用户ID
	}

	err := h.adService.DeptGroupMapping.UpdateMapping(c.Request.Context(), id, updateReq)
	if err != nil {
		response.Error(c, http.StatusInternalServerError, "更新失败: "+err.Error())
		return
	}

	response.Success(c, nil)
}

// AutoMapDepartments 批量自动映射部门
// @Summary 批量自动映射部门到AD组
// @Description 根据部门名称自动匹配对应的AD组（cxhub-{dept}命名规则）
// @Tags AD Group Sync
// @Accept json
// @Produce json
// @Param request body AutoMapRequest true "自动映射请求"
// @Success 200 {object} response.Response "自动映射成功"
// @Router /api/v1/ad/groups/automap [post]
func (h *GroupSyncHandler) AutoMapDepartments(c *gin.Context) {
	var req AutoMapRequest
	if err := c.ShouldBindJSON(&req); err != nil {
		response.Error(c, http.StatusBadRequest, "请求参数错误: "+err.Error())
		return
	}

	var result interface{}
	var err error

	if len(req.DeptIDs) == 0 {
		// 自动映射所有二级部门
		result, err = h.adService.DeptGroupMapping.AutoMapAllDepartments(c.Request.Context(), req.ConfigID)
	} else {
		// 映射指定部门（TODO: 实现批量指定部门映射）
		err = h.adService.DeptGroupMapping.AutoMapDepartments(c.Request.Context(), req.ConfigID, req.DeptIDs)
		result = map[string]interface{}{
			"message": "批量映射完成",
			"count":   len(req.DeptIDs),
		}
	}

	if err != nil {
		response.Error(c, http.StatusInternalServerError, "自动映射失败: "+err.Error())
		return
	}

	response.Success(c, result)
}
```

```go
package addomain

import (
	"github.com/gin-gonic/gin"
	"github.com/xingran-next/xingran-go-backend/internal/services/addomain"
	"github.com/xingran-next/xingran-go-backend/pkg/middleware"
)

// SetupGroupSyncRouter 设置组同步API路由
func SetupGroupSyncRouter(r *gin.RouterGroup, adService *addomain.ADDomainService) {
	handler := NewGroupSyncHandler(adService)

	// 应用认证中间件
	auth := r.Group("")
	// auth.Use(middleware.Auth()) // TODO: 根据项目需求启用认证

	group := auth.Group("/ad/groups")
	{
		// 同步操作
		group.POST("/sync", handler.SyncAllMembers)                      // 手动触发同步所有成员
		group.GET("/sync/status", handler.GetSyncStatus)                 // 查询同步状态
		group.POST("/sync/dept/:deptId", handler.SyncDeptMembers)        // 同步指定部门
		group.POST("/exclusive", handler.EnsureExclusiveMembership)      // 排他性成员关系

		// 映射管理
		group.GET("/mappings", handler.ListMappings)          // 查询映射列表
		group.POST("/mappings", handler.GetMapping)           // 创建映射
		group.GET("/mappings/:id", handler.GetMapping)        // 获取映射详情
		group.PUT("/mappings/:id", handler.UpdateMapping)     // 更新映射
		group.DELETE("/mappings/:id", handler.DeleteMapping)  // 删除映射

		// 自动映射
		group.POST("/automap", handler.AutoMapDepartments)    // 批量自动映射
	}
}
```

**Router设计要点**:
1. **逻辑分组**: 所有组同步端点在/api/v1/ad/groups下
2. **RESTful URLs**: 遵循REST conventions
3. **中间件钩子**: 注释指示认证/权限应该应用的位置
4. **一致模式**: 遵循现有router结构
