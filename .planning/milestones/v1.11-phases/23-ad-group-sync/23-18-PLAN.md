# Plan 23-18: 前端UI - 批量自动映射页面

## Objective
创建批量自动映射页面，支持查看可映射部门、预览映射结果、执行批量自动映射和进度监控，简化部门到AD组的映射配置流程。

**Output**: 完整的自动映射页面，包括部门列表、映射预览、批量操作和进度反馈。

```typescript
import type {
  MappableDept,
  AutoMappingPreview,
  AutoMappingResult
} from '@/types/ad-group';

// ===== AD组自动映射 API =====

/**
 * 获取可映射的二级部门列表
 */
export const getMappableDepts = async (configId: string): Promise<MappableDept[]> => {
  const result = await get<MappableDept[]>('/ad/groups/mappable-depts', {
    params: { configId }
  });
  return result.data;
};

/**
 * 预览自动映射结果
 */
export const previewAutoMapping = async (configId: string, deptIds?: string[]): Promise<AutoMappingPreview> => {
  const result = await post<AutoMappingPreview>('/ad/groups/automap/preview', {
    configId,
    deptIds
  });
  return result.data;
};

/**
 * 批量自动映射部门
 */
export const autoMapDepartments = async (configId: string, deptIds?: string[]): Promise<AutoMappingResult> => {
  const result = await post<AutoMappingResult>('/ad/groups/automap', {
    configId,
    deptIds
  });
  return result.data;
};

/**
 * 一键自动映射所有二级部门
 */
export const autoMapAllDepartments = async (configId: string): Promise<AutoMappingResult> => {
  const result = await post<AutoMappingResult>('/ad/groups/automap-all', {
    configId
  });
  return result.data;
};
```

**API设计要点**:
1. 获取可映射部门列表
2. 预览映射结果
3. 执行批量映射
4. 一键映射所有部门
5. 与后端API一一对应
</action>

```typescript
import React, { useState, useEffect, useCallback } from 'react';
import {
  Card,
  Table,
  Button,
  Space,
  Select,
  Modal,
  Alert,
  Progress,
  Tag,
  message,
  Tooltip,
  Statistic,
  Row,
  Col,
  Descriptions,
  Steps
} from 'antd';
import {
  CheckCircleOutlined,
  CloseCircleOutlined,
  InfoCircleOutlined,
  ThunderboltOutlined,
  EyeOutlined,
  SyncOutlined
} from '@ant-design/icons';
import type { ColumnsType } from 'antd/es/table';
import {
  getMappableDepts,
  previewAutoMapping,
  autoMapDepartments,
  autoMapAllDepartments
} from '@/lib/api';
import type {
  MappableDept,
  AutoMappingPreview,
  AutoMappingResult
} from '@/types/ad-group';

const { Step } = Steps;
const { Option } = Select;

const AutoMapping: React.FC = () => {
  // 状态管理
  const [depts, setDepts] = useState<MappableDept[]>([]);
  const [loading, setLoading] = useState(false);
  const [mapping, setMapping] = useState(false);
  const [currentConfig, setCurrentConfig] = useState<string>('');
  const [selectedDepts, setSelectedDepts] = useState<string[]>([]);
  const [preview, setPreview] = useState<AutoMappingPreview | null>(null);
  const [result, setResult] = useState<AutoMappingResult | null>(null);
  const [currentStep, setCurrentStep] = useState(0);

  // 加载可映射部门
  const loadMappableDepts = useCallback(async () => {
    if (!currentConfig) {
      message.warning('请先选择AD配置');
      return;
    }

    setLoading(true);
    try {
      const data = await getMappableDepts(currentConfig);
      setDepts(data);
      setCurrentStep(1);
    } catch (error) {
      message.error('加载部门列表失败');
    } finally {
      setLoading(false);
    }
  }, [currentConfig]);

  // 预览映射
  const handlePreview = async () => {
    if (selectedDepts.length === 0) {
      message.warning('请先选择要映射的部门');
      return;
    }

    setLoading(true);
    try {
      const data = await previewAutoMapping(currentConfig, selectedDepts);
      setPreview(data);
      setCurrentStep(2);
    } catch (error) {
      message.error('预览失败');
    } finally {
      setLoading(false);
    }
  };

  // 执行映射
  const handleMap = async () => {
    setLoading(true);
    setMapping(true);
    try {
      const data = await autoMapDepartments(currentConfig, selectedDepts);
      setResult(data);
      setCurrentStep(3);

      if (data.success) {
        message.success(`映射完成: ${data.createdCount}创建, ${data.updatedCount}更新`);
      } else {
        message.warning(`映射完成: ${data.failedCount}失败`);
      }

      // 重新加载部门列表
      loadMappableDepts();
    } catch (error: any) {
      message.error(error.message || '映射失败');
    } finally {
      setLoading(false);
      setMapping(false);
    }
  };

  // 一键映射所有部门
  const handleMapAll = async () => {
    if (!currentConfig) {
      message.warning('请先选择AD配置');
      return;
    }

    Modal.confirm({
      title: '确认一键映射',
      content: `将自动映射所有二级部门到对应的AD组（cxhub-{部门名}），确定继续？`,
      onOk: async () => {
        setLoading(true);
        setMapping(true);
        try {
          const data = await autoMapAllDepartments(currentConfig);
          setResult(data);
          setCurrentStep(3);

          if (data.success) {
            message.success(`映射完成: ${data.createdCount}创建, ${data.updatedCount}更新`);
          } else {
            message.warning(`映射完成: ${data.failedCount}失败`);
          }

          loadMappableDepts();
        } catch (error: any) {
          message.error(error.message || '映射失败');
        } finally {
          setLoading(false);
          setMapping(false);
        }
      }
    });
  };

  // 重置流程
  const handleReset = () => {
    setCurrentStep(0);
    setSelectedDepts([]);
    setPreview(null);
    setResult(null);
  };

  // 表格列定义
  const columns: ColumnsType<MappableDept> = [
    {
      title: '部门名称',
      dataIndex: 'deptName',
      key: 'deptName',
      width: 200
    },
    {
      title: '建议组名',
      dataIndex: 'suggestedGroupName',
      key: 'suggestedGroupName',
      width: 200,
      render: (name: string, record) => (
        <Space>
          <span>{name}</span>
          {record.groupExists ? (
            <Tag color="success">组存在</Tag>
          ) : (
            <Tag color="warning">需创建</Tag>
          )}
        </Space>
      )
    },
    {
      title: '组DN',
      dataIndex: 'suggestedGroupDN',
      key: 'suggestedGroupDN',
      ellipsis: true,
      render: (dn?: string) => dn || '-'
    },
    {
      title: '当前映射',
      key: 'mapping',
      width: 150,
      render: (_, record) => {
        if (record.hasMapping && record.existingMapping) {
          return (
            <Tooltip title={`组: ${record.existingMapping.adGroupName}`}>
              <Tag color={record.existingMapping.syncEnabled ? 'success' : 'default'}>
                已映射
              </Tag>
            </Tooltip>
          );
        }
        return <Tag color="default">未映射</Tag>;
      }
    },
    {
      title: '操作',
      key: 'action',
      width: 100,
      render: (_, record) => {
        if (record.hasMapping) {
          return (
            <Button
              type="link"
              size="small"
              disabled
            >
              已映射
            </Button>
          );
        }
        return (
          <Button
            type="link"
            size="small"
            onClick={() => setSelectedDepts([...selectedDepts, record.id])}
          >
            选择
          </Button>
        );
      }
    }
  ];

  // 渲染步骤指示器
  const renderSteps = () => (
    <Card style={{ marginBottom: 24 }}>
      <Steps current={currentStep}>
        <Step title="选择配置" description="选择AD配置" />
        <Step title="选择部门" description="选择要映射的部门" />
        <Step title="预览映射" description="预览映射结果" />
        <Step title="执行映射" description="执行自动映射" />
      </Steps>
    </Card>
  );

  // 渲染配置选择
  const renderConfigSelect = () => (
    <Card>
      <Descriptions column={1} bordered>
        <Descriptions.Item label="AD配置">
          <Select
            style={{ width: 300 }}
            placeholder="选择AD配置"
            value={currentConfig}
            onChange={setCurrentConfig}
          >
            {/* TODO: 从API加载AD配置列表 */}
            <Option value="config-1">AD配置1</Option>
          </Select>
        </Descriptions.Item>
        <Descriptions.Item label="操作">
          <Button
            type="primary"
            icon={<SyncOutlined />}
            onClick={loadMappableDepts}
            disabled={!currentConfig}
            loading={loading}
          >
            加载可映射部门
          </Button>
        </Descriptions.Item>
      </Descriptions>
    </Card>
  );

  // 渲染部门选择
  const renderDeptSelection = () => (
    <Card>
      <Row gutter={16} style={{ marginBottom: 16 }}>
        <Col span={18}>
          <Space>
            <span>已选择: {selectedDepts.length} 个部门</span>
            <Button onClick={() => setSelectedDepts([])}>清空选择</Button>
          </Space>
        </Col>
        <Col span={6} style={{ textAlign: 'right' }}>
          <Space>
            <Button onClick={handlePreview} disabled={selectedDepts.length === 0}>
              预览映射
            </Button>
            <Button
              type="primary"
              icon={<ThunderboltOutlined />}
              onClick={handleMapAll}
              loading={mapping}
            >
              一键映射所有
            </Button>
          </Space>
        </Col>
      </Row>

      <Table
        columns={columns}
        dataSource={depts}
        rowKey="id"
        loading={loading}
        rowSelection={{
          selectedRowKeys: selectedDepts,
          onChange: (keys) => setSelectedDepts(keys as string[]),
          getCheckboxProps: (record) => ({
            disabled: record.hasMapping
          })
        }}
        pagination={{
          showSizeChanger: true,
          showTotal: (t) => `共 ${t} 条`
        }}
      />
    </Card>
  );

  // 渲染预览结果
  const renderPreview = () => {
    if (!preview) return null;

    return (
      <Card>
        <Row gutter={16} style={{ marginBottom: 16 }}>
          <Col span={6}>
            <Statistic title="总部门数" value={preview.totalDepts} />
          </Col>
          <Col span={6}>
            <Statistic
              title="可映射"
              value={preview.canMapCount}
              valueStyle={{ color: '#3f8600' }}
            />
          </Col>
          <Col span={6}>
            <Statistic
              title="已映射"
              value={preview.alreadyMappedCount}
              valueStyle={{ color: '#1890ff' }}
            />
          </Col>
          <Col span={6}>
            <Statistic
              title="组不存在"
              value={preview.groupNotFoundCount}
              valueStyle={{ color: '#cf1322' }}
            />
          </Col>
        </Row>

        <Alert
          message="映射预览"
          description={
            <ul>
              {preview.mappings.map((m, i) => (
                <li key={i}>
                  {m.deptName} → {m.groupName}
                  <Tag color={m.action === 'create' ? 'blue' : m.action === 'update' ? 'green' : 'default'}>
                    {m.action === 'create' ? '创建' : m.action === 'update' ? '更新' : '跳过'}
                  </Tag>
                  {m.reason && <span style={{ color: '#999' }}> ({m.reason})</span>}
                </li>
              ))}
            </ul>
          }
          type="info"
          showIcon
        />

        <div style={{ marginTop: 16, textAlign: 'right' }}>
          <Space>
            <Button onClick={() => setCurrentStep(1)}>返回</Button>
            <Button
              type="primary"
              onClick={handleMap}
              loading={mapping}
              disabled={preview.canMapCount === 0}
            >
              执行映射
            </Button>
          </Space>
        </div>
      </Card>
    );
  };

  // 渲染执行结果
  const renderResult = () => {
    if (!result) return null;

    return (
      <Card>
        {result.success ? (
          <Alert
            message="映射完成"
            type="success"
            showIcon
            style={{ marginBottom: 16 }}
          />
        ) : (
          <Alert
            message="映射完成（有失败）"
            type="warning"
            showIcon
            style={{ marginBottom: 16 }}
          />
        )}

        <Row gutter={16} style={{ marginBottom: 16 }}>
          <Col span={6}>
            <Statistic title="处理总数" value={result.totalProcessed} />
          </Col>
          <Col span={6}>
            <Statistic
              title="创建映射"
              value={result.createdCount}
              valueStyle={{ color: '#3f8600' }}
            />
          </Col>
          <Col span={6}>
            <Statistic
              title="更新映射"
              value={result.updatedCount}
              valueStyle={{ color: '#1890ff' }}
            />
          </Col>
          <Col span={6}>
            <Statistic
              title="失败"
              value={result.failedCount}
              valueStyle={{ color: '#cf1322' }}
            />
          </Col>
        </Row>

        {result.errors.length > 0 && (
          <Alert
            message="错误详情"
            description={
              <ul>
                {result.errors.map((err, i) => (
                  <li key={i}>
                    {err.deptName}: {err.error}
                  </li>
                ))}
              </ul>
            }
            type="error"
            showIcon
          />
        )}

        <div style={{ marginTop: 16, textAlign: 'right' }}>
          <Space>
            <Button onClick={handleReset}>重新开始</Button>
            <Button type="primary" onClick={() => setCurrentStep(1)}>
              继续映射
            </Button>
          </Space>
        </div>
      </Card>
    );
  };

  return (
    <div style={{ padding: 24 }}>
      <Card
        title="批量自动映射"
        extra={
          <Button onClick={handleReset} disabled={currentStep === 0}>
            重新开始
          </Button>
        }
      >
        {renderSteps()}

        {currentStep === 0 && renderConfigSelect()}
        {currentStep === 1 && renderDeptSelection()}
        {currentStep === 2 && renderPreview()}
        {currentStep === 3 && renderResult()}
      </Card>
    </div>
  );
};

export default AutoMapping;
```

```typescript
const usageGuide = (
  <Alert
    message="使用说明"
    description={
```
