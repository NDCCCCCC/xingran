```go
package addomain

const (
	// ConfigGroupSyncEnabled 组同步开关
	ConfigGroupSyncEnabled = "sys.ad.group.sync.enabled"
	// ConfigGroupSyncCron 组同步Cron表达式
	ConfigGroupSyncCron = "sys.ad.group.sync.cron"
	// ConfigGroupMemberOU MemberOU路径
	ConfigGroupMemberOU = "sys.ad.group.member_ou"
	// ConfigGroupAutoCreate 自动创建组
	ConfigGroupAutoCreate = "sys.ad.group.auto_create"
	// ConfigGroupMaxConcurrent 最大并发数
	ConfigGroupMaxConcurrent = "sys.ad.group.max_concurrent"
	// ConfigGroupSyncBatchSize 批量同步大小
	ConfigGroupSyncBatchSize = "sys.ad.group.sync.batch_size"
)

// GroupSyncConfig 组同步配置
type GroupSyncConfig struct {
	Enabled          bool   `json:"enabled"`
	Cron             string `json:"cron"`
	MemberOU         string `json:"member_ou"`
	AutoCreateGroups bool   `json:"auto_create_groups"`
	MaxConcurrent    int    `json:"max_concurrent"`
	SyncBatchSize    int    `json:"sync_batch_size"`
}

// GetDefaultGroupSyncConfig 获取默认配置
func GetDefaultGroupSyncConfig() *GroupSyncConfig {
	return &GroupSyncConfig{
		Enabled:          false,
		Cron:             "0 */15 * * * *",
		MemberOU:         "",
		AutoCreateGroups: true,
		MaxConcurrent:    5,
		SyncBatchSize:    100,
	}
}
```

**配置参数说明**:
1. `sys.ad.group.sync.enabled`: 全局开关，控制是否执行组同步
2. `sys.ad.group.sync.cron`: Cron表达式，控制同步频率
3. `sys.ad.group.member_ou`: 指定部门组的目标OU（留空=使用AD配置的OU）
4. `sys.ad.group.auto_create`: 是否自动创建不存在的组
5. `sys.ad.group.max_concurrent`: 最大并发同步数（防止LDAP连接耗尽）
6. `sys.ad.group.sync.batch_size`: 批量处理的成员数量

**默认值设计理由**:
- `enabled: false`: 默认禁用，避免未配置时自动执行
- `cron: "0 */15 * * * *"`: 15分钟同步一次，平衡实时性和性能
- `auto_create: true`: 自动创建组，减少手动配置
- `max_concurrent: 5`: 限制并发数，避免LDAP服务器压力

```go
package addomain

import (
	"context"
	"fmt"
	"strconv"
)

// GroupConfigService 组配置服务
type GroupConfigService struct {
	configService ConfigProvider
}

// ConfigProvider 配置提供者接口
type ConfigProvider interface {
	GetConfig(ctx context.Context, configKey string) (string, error)
	SetConfig(ctx context.Context, configKey string, configValue string) error
}

// NewGroupConfigService 创建组配置服务
func NewGroupConfigService(configProvider ConfigProvider) *GroupConfigService {
	return &GroupConfigService{
		configService: configProvider,
	}
}

// GetGroupSyncConfig 获取组同步配置
func (s *GroupConfigService) GetGroupSyncConfig(ctx context.Context) (*GroupSyncConfig, error) {
	config := GetDefaultGroupSyncConfig()

	// 读取enabled
	enabledStr, err := s.configService.GetConfig(ctx, ConfigGroupSyncEnabled)
	if err == nil && enabledStr != "" {
		config.Enabled = enabledStr == "true" || enabledStr == "1"
	}

	// 读取cron
	cronStr, err := s.configService.GetConfig(ctx, ConfigGroupSyncCron)
	if err == nil && cronStr != "" {
		config.Cron = cronStr
	}

	// 读取member_ou
	memberOU, err := s.configService.GetConfig(ctx, ConfigGroupMemberOU)
	if err == nil && memberOU != "" {
		config.MemberOU = memberOU
	}

	// 读取auto_create
	autoCreateStr, err := s.configService.GetConfig(ctx, ConfigGroupAutoCreate)
	if err == nil && autoCreateStr != "" {
		config.AutoCreateGroups = autoCreateStr == "true" || autoCreateStr == "1"
	}

	// 读取max_concurrent
	maxConcurrentStr, err := s.configService.GetConfig(ctx, ConfigGroupMaxConcurrent)
	if err == nil && maxConcurrentStr != "" {
		if val, err := strconv.Atoi(maxConcurrentStr); err == nil {
			config.MaxConcurrent = val
		}
	}

	// 读取batch_size
	batchSizeStr, err := s.configService.GetConfig(ctx, ConfigGroupSyncBatchSize)
	if err == nil && batchSizeStr != "" {
		if val, err := strconv.Atoi(batchSizeStr); err == nil {
			config.SyncBatchSize = val
		}
	}

	return config, nil
}

// IsGroupSyncEnabled 检查组同步是否启用
func (s *GroupConfigService) IsGroupSyncEnabled(ctx context.Context) bool {
	config, err := s.GetGroupSyncConfig(ctx)
	if err != nil {
		return false
	}
	return config.Enabled
}

// UpdateGroupSyncConfig 更新组同步配置
func (s *GroupConfigService) UpdateGroupSyncConfig(ctx context.Context, config *GroupSyncConfig) error {
	// 更新enabled
	if err := s.configService.SetConfig(ctx, ConfigGroupSyncEnabled, strconv.FormatBool(config.Enabled)); err != nil {
		return fmt.Errorf("failed to update enabled: %w", err)
	}

	// 更新cron
	if err := s.configService.SetConfig(ctx, ConfigGroupSyncCron, config.Cron); err != nil {
		return fmt.Errorf("failed to update cron: %w", err)
	}

	// 更新member_ou
	if err := s.configService.SetConfig(ctx, ConfigGroupMemberOU, config.MemberOU); err != nil {
		return fmt.Errorf("failed to update member_ou: %w", err)
	}

	// 更新auto_create
	if err := s.configService.SetConfig(ctx, ConfigGroupAutoCreate, strconv.FormatBool(config.AutoCreateGroups)); err != nil {
		return fmt.Errorf("failed to update auto_create: %w", err)
	}

	// 更新max_concurrent
	if err := s.configService.SetConfig(ctx, ConfigGroupMaxConcurrent, strconv.Itoa(config.MaxConcurrent)); err != nil {
		return fmt.Errorf("failed to update max_concurrent: %w", err)
	}

	// 更新batch_size
	if err := s.configService.SetConfig(ctx, ConfigGroupSyncBatchSize, strconv.Itoa(config.SyncBatchSize)); err != nil {
		return fmt.Errorf("failed to update batch_size: %w", err)
	}

	return nil
}

// ValidateConfig 验证配置有效性
func (s *GroupConfigService) ValidateConfig(config *GroupSyncConfig) error {
	// 验证cron表达式
	if config.Cron == "" {
		return fmt.Errorf("cron表达式不能为空")
	}

	// 验证并发数
	if config.MaxConcurrent < 1 || config.MaxConcurrent > 20 {
		return fmt.Errorf("max_concurrent必须在1-20之间")
	}

	// 验证批量大小
	if config.SyncBatchSize < 10 || config.SyncBatchSize > 1000 {
		return fmt.Errorf("sync_batch_size必须在10-1000之间")
	}

	return nil
}
```

**设计要点**:
1. 使用接口 `ConfigProvider` 解耦配置服务
2. 提供默认值机制，确保配置缺失时不影响功能
3. 实现配置验证，防止无效配置
4. 支持运行时动态更新，无需重启服务
</action>
