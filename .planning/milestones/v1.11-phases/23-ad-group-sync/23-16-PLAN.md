```typescript
import { post, get, put, del } from './api';
import type {
  DeptGroupMapping,
  GroupMappingListResponse,
  CreateGroupMappingRequest,
  UpdateGroupMappingRequest,
  ListGroupMappingsParams,
  SyncResult
} from '@/types/ad-group';

// ===== AD组映射管理 API =====

/**
 * 查询部门-组映射列表
 */
export const getGroupMappings = async (params: ListGroupMappingsParams = {}): Promise<GroupMappingListResponse> => {
  const result = await post<GroupMappingListResponse>('/ad/groups/mappings', params);
  return result.data;
};

/**
 * 获取单个映射详情
 */
export const getGroupMapping = async (id: string): Promise<DeptGroupMapping> => {
  const result = await get<DeptGroupMapping>(`/ad/groups/mappings/${id}`);
  return result.data;
};

/**
 * 创建部门-组映射
 */
export const createGroupMapping = async (data: CreateGroupMappingRequest): Promise<DeptGroupMapping> => {
  const result = await post<DeptGroupMapping>('/ad/groups/mappings', data);
  return result.data;
};

/**
 * 更新部门-组映射
 */
export const updateGroupMapping = async (id: string, data: UpdateGroupMappingRequest): Promise<void> => {
  await put<void>(`/ad/groups/mappings/${id}`, data);
};

/**
 * 删除部门-组映射
 */
export const deleteGroupMapping = async (id: string): Promise<void> => {
  await del<void>(`/ad/groups/mappings/${id}`);
};

/**
 * 批量自动映射部门
 */
export const autoMapDepartments = async (configId: string, deptIds?: string[]): Promise<SyncResult> => {
  const result = await post<SyncResult>('/ad/groups/automap', {
    configId,
    deptIds
  });
  return result.data;
};

/**
 * 手动触发同步所有成员
 */
export const syncAllMembers = async (configId: string): Promise<SyncResult> => {
  const result = await post<SyncResult>('/ad/groups/sync', { configId });
  return result.data;
};

/**
 * 同步指定部门的成员
 */
export const syncDeptMembers = async (deptId: string): Promise<SyncResult> => {
  const result = await post<SyncResult>(`/ad/groups/sync/dept/${deptId}`);
  return result.data;
};

/**
 * 确保排他性成员关系
 */
export const ensureExclusiveMembership = async (configId: string): Promise<SyncResult> => {
  const result = await post<SyncResult>('/ad/groups/exclusive', null, {
    params: { configId }
  });
  return result.data;
};

/**
 * 查询同步状态
 */
export const getSyncStatus = async (configId?: string): Promise<any> => {
  const result = await get<any>('/ad/groups/sync/status', {
    params: configId ? { configId } : undefined
  });
  return result.data;
};
```

2. 完整的类型定义
3. 清晰的函数命名
4. 与后端API端点一一对应

```typescript
import React, { useState, useEffect, useCallback } from 'react';
import {
  Table,
  Button,
  Space,
  Modal,
  Form,
  Select,
  Switch,
  message,
  Popconfirm,
  Tag,
  Tooltip,
  Card,
  Statistic,
  Row,
  Col
} from 'antd';
import {
  PlusOutlined,
  EditOutlined,
  DeleteOutlined,
  SyncOutlined,
  ReloadOutlined,
  AutoFixHighOutlined
} from '@ant-design/icons';
import type { ColumnsType } from 'antd/es/table';
import {
  getGroupMappings,
  createGroupMapping,
  updateGroupMapping,
  deleteGroupMapping,
  syncDeptMembers,
  getSyncStatus
} from '@/lib/api';
import type {
  DeptGroupMapping,
  CreateGroupMappingRequest,
  UpdateGroupMappingRequest
} from '@/types/ad-group';

const { Option } = Select;

const GroupMapping: React.FC = () => {
  // 状态管理
  const [mappings, setMappings] = useState<DeptGroupMapping[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(false);
  const [current, setCurrent] = useState(1);
  const [pageSize, setPageSize] = useState(10);

  // 对话框状态
  const [createModalVisible, setCreateModalVisible] = useState(false);
  const [editModalVisible, setEditModalVisible] = useState(false);
  const [currentMapping, setCurrentMapping] = useState<DeptGroupMapping | null>(null);
  const [form] = Form.useForm();
  const [editForm] = Form.useForm();

  // 同步状态
  const [syncingDepts, setSyncingDepts] = useState<Set<string>>(new Set());

  // 加载映射列表
  const loadMappings = useCallback(async () => {
    setLoading(true);
    try {
      const response = await getGroupMappings({ current, pageSize });
      setMappings(response.list);
      setTotal(response.total);
    } catch (error) {
      message.error('加载映射列表失败');
    } finally {
      setLoading(false);
    }
  }, [current, pageSize]);

  // 初始化加载
  useEffect(() => {
    loadMappings();
  }, [loadMappings]);

  // 创建映射
  const handleCreate = async (values: CreateGroupMappingRequest) => {
    try {
      await createGroupMapping(values);
      message.success('创建映射成功');
      setCreateModalVisible(false);
      form.resetFields();
      loadMappings();
    } catch (error: any) {
      message.error(error.message || '创建映射失败');
    }
  };

  // 更新映射
  const handleUpdate = async (values: UpdateGroupMappingRequest) => {
    if (!currentMapping) return;

    try {
      await updateGroupMapping(currentMapping.id, values);
      message.success('更新映射成功');
      setEditModalVisible(false);
      editForm.resetFields();
      setCurrentMapping(null);
      loadMappings();
    } catch (error: any) {
      message.error(error.message || '更新映射失败');
    }
  };

  // 删除映射
  const handleDelete = async (id: string) => {
    try {
      await deleteGroupMapping(id);
      message.success('删除映射成功');
      loadMappings();
    } catch (error: any) {
      message.error(error.message || '删除映射失败');
    }
  };

  // 同步部门成员
  const handleSync = async (deptId: string) => {
    setSyncingDepts(prev => new Set(prev).add(deptId));
    try {
      const result = await syncDeptMembers(deptId);
      if (result.success) {
        message.success(`同步成功: ${result.successCount}/${result.totalProcessed}`);
      } else {
        message.warning(`同步完成: ${result.successCount}成功, ${result.failedCount}失败`);
      }
      loadMappings();
    } catch (error: any) {
      message.error(error.message || '同步失败');
    } finally {
      setSyncingDepts(prev => {
        const newSet = new Set(prev);
        newSet.delete(deptId);
        return newSet;
      });
    }
  };

  // 打开编辑对话框
  const openEditModal = (mapping: DeptGroupMapping) => {
    setCurrentMapping(mapping);
    editForm.setFieldsValue({
      mappingStatus: mapping.mappingStatus,
      syncEnabled: mapping.syncEnabled
    });
    setEditModalVisible(true);
  };

  // 表格列定义
  const columns: ColumnsType<DeptGroupMapping> = [
    {
      title: '部门',
      dataIndex: 'deptName',
      key: 'deptName',
      width: 150
    },
    {
      title: 'AD组',
      dataIndex: 'adGroupName',
      key: 'adGroupName',
      width: 200,
      render: (text, record) => (
        <div>
          <div>{text}</div>
          <div style={{ fontSize: 12, color: '#999' }}>{record.adGroupDN}</div>
        </div>
      )
    },
    {
      title: '映射类型',
      dataIndex: 'mappingType',
      key: 'mappingType',
      width: 100,
      render: (type: string) => (
        <Tag color={type === 'auto' ? 'blue' : 'green'}>
          {type === 'auto' ? '自动' : '手动'}
        </Tag>
      )
    },
    {
      title: '成员数',
      dataIndex: 'memberCount',
      key: 'memberCount',
      width: 100,
      render: (count: number) => <Statistic value={count} valueStyle={{ fontSize: 14 }} />
    },
    {
      title: '同步状态',
      dataIndex: 'syncEnabled',
      key: 'syncEnabled',
      width: 100,
      render: (enabled: boolean) => (
        <Tag color={enabled ? 'success' : 'default'}>
          {enabled ? '已启用' : '已禁用'}
        </Tag>
      )
    },
    {
      title: '最后同步',
      dataIndex: 'lastSyncAt',
      key: 'lastSyncAt',
      width: 180,
      render: (time: string) => time ? new Date(time).toLocaleString('zh-CN') : '-'
    },
    {
      title: '操作',
      key: 'action',
      width: 250,
      fixed: 'right',
      render: (_, record) => (
        <Space size="small">
          <Tooltip title="同步成员">
            <Button
              type="link"
              icon={<SyncOutlined />}
              loading={syncingDepts.has(record.deptId)}
              onClick={() => handleSync(record.deptId)}
              disabled={!record.syncEnabled}
            />
          </Tooltip>
          <Tooltip title="编辑">
            <Button
              type="link"
              icon={<EditOutlined />}
              onClick={() => openEditModal(record)}
            />
          </Tooltip>
          <Popconfirm
            title="确定删除此映射？"
            onConfirm={() => handleDelete(record.id)}
            okText="确定"
            cancelText="取消"
          >
            <Button type="link" danger icon={<DeleteOutlined />} />
          </Popconfirm>
        </Space>
      )
    }
  ];

  return (
    <div style={{ padding: 24 }}>
      <Card>
        <Row gutter={16} style={{ marginBottom: 16 }}>
          <Col span={6}>
            <Statistic title="总映射数" value={total} />
          </Col>
          <Col span={6}>
            <Statistic
              title="启用同步"
              value={mappings.filter(m => m.syncEnabled).length}
              valueStyle={{ color: '#3f8600' }}
            />
          </Col>
          <Col span={6}>
            <Statistic
              title="自动映射"
              value={mappings.filter(m => m.mappingType === 'auto').length}
              valueStyle={{ color: '#1890ff' }}
            />
          </Col>
          <Col span={6}>
            <Button
              type="primary"
              icon={<ReloadOutlined />}
              onClick={loadMappings}
              loading={loading}
            >
              刷新
            </Button>
          </Col>
        </Row>

        <div style={{ marginBottom: 16 }}>
          <Space>
            <Button
              type="primary"
              icon={<PlusOutlined />}
              onClick={() => setCreateModalVisible(true)}
            >
              创建映射
            </Button>
          </Space>
        </div>

        <Table
          columns={columns}
          dataSource={mappings}
          rowKey="id"
          loading={loading}
          scroll={{ x: 1200 }}
          pagination={{
            current,
            pageSize,
            total,
            showSizeChanger: true,
            showTotal: (t) => `共 ${t} 条`,
            onChange: (page, size) => {
              setCurrent(page);
              setPageSize(size || 10);
            }
          }}
        />
      </Card>

      {/* 创建映射对话框 */}
      <Modal
        title="创建部门-组映射"
        open={createModalVisible}
        onCancel={() => {
          setCreateModalVisible(false);
          form.resetFields();
        }}
        onOk={() => form.submit()}
        width={600}
      >
        <Form
          form={form}
          layout="vertical"
          onFinish={handleCreate}
        >
          <Form.Item
            label="AD配置"
            name="adConfigId"
            rules={[{ required: true, message: '请选择AD配置' }]}
          >
            <Select placeholder="请选择AD配置">
              {/* TODO: 从API加载AD配置列表 */}
              <Option value="config-1">AD配置1</Option>
            </Select>
          </Form.Item>

          <Form.Item
            label="部门"
            name="deptId"
            rules={[{ required: true, message: '请选择部门' }]}
          >
            <Select placeholder="请选择部门">
              {/* TODO: 从API加载部门列表 */}
              <Option value="dept-1">科技创新部</Option>
            </Select>
          </Form.Item>

          <Form.Item
            label="AD组"
            name="adGroupId"
            rules={[{ required: true, message: '请选择AD组' }]}
          >
            <Select placeholder="请选择AD组">
              {/* TODO: 从API加载AD组列表 */}
              <Option value="group-1">cxhub-科技创新</Option>
            </Select>
          </Form.Item>

          <Form.Item
            label="映射类型"
            name="mappingType"
            initialValue="manual"
          >
            <Select>
              <Option value="manual">手动</Option>
              <Option value="auto">自动</Option>
            </Select>
          </Form.Item>

          <Form.Item
            label="启用同步"
            name="syncEnabled"
            valuePropName="checked"
            initialValue={true}
          >
            <Switch />
          </Form.Item>
        </Form>
      </Modal>

      {/* 编辑映射对话框 */}
      <Modal
        title="编辑映射"
        open={editModalVisible}
        onCancel={() => {
          setEditModalVisible(false);
          editForm.resetFields();
          setCurrentMapping(null);
        }}
        onOk={() => editForm.submit()}
        width={500}
      >
        <Form
          form={editForm}
          layout="vertical"
          onFinish={handleUpdate}
        >
          <Form.Item
            label="映射状态"
            name="mappingStatus"
            rules={[{ required: true, message: '请选择映射状态' }]}
          >
            <Select>
              <Option value="active">活动</Option>

          <Form.Item
            label="启用同步"
            name="syncEnabled"
            valuePropName="checked"
          >
            <Switch />
          </Form.Item>
        </Form>
      </Modal>
    </div>
  );
};

export default GroupMapping;
```
