### Task 1: 实现 Agent PTY 管理器（完整）

**Files**: `internal/agent/server/pty_manager.go`

**Action**:
```go
package server

import (
   "context"
   "os"
   "os/exec"
   "sync"
   
   "github.com/creack/pty"
)

type PTYManager struct {
   sessions map[string]*PTYSession
   mu       sync.RWMutex
}

type PTYSession struct {
   ID     string
   Pty    *os.File
   Cmd    *exec.Cmd
   Input  io.Writer
   Output io.Reader
   mu     sync.Mutex
}

func NewPTYManager() *PTYManager {
   return &PTYManager{
      sessions: make(map[string]*PTYSession),
   }
}

func (m *PTYManager) CreateSession(ctx context.Context, sessionID, shell string) (*PTYSession, error) {
   m.mu.Lock()
   defer m.mu.Unlock()

   // 确定默认 Shell
   if shell == "" {
      if runtime.GOOS == "windows" {
         shell = "powershell.exe"
      } else {
         shell = "/bin/bash"
      }
   }

   // 创建伪终端和命令
   cmd := exec.Command(shell)
   ptyFile, err := pty.Start(cmd)
   if err != nil {
      return nil, err
   }

   session := &PTYSession{
      ID:     sessionID,
      Pty:    ptyFile,
      Cmd:    cmd,
      Input:  ptyFile,
      Output: ptyFile,
   }

   m.sessions[sessionID] = session
   return session, nil
}

func (m *PTYManager) CloseSession(sessionID string) error {
   m.mu.Lock()
   defer m.mu.Unlock()

   session, exists := m.sessions[sessionID]
   if !exists {
      return nil
   }

   // 关闭 PTY 和进程
   session.Pty.Close()
   if session.Cmd.Process != nil {
      session.Cmd.Process.Kill()
   }

   delete(m.sessions, sessionID)
   return nil
}

func (m *PTYManager) WriteToSession(sessionID string, data []byte) error {
   m.mu.RLock()
   session, exists := m.sessions[sessionID]
   m.mu.RUnlock()

   if !exists {
      return fmt.Errorf("session not found")
   }

   session.mu.Lock()
   defer session.mu.Unlock()
   _, err := session.Input.Write(data)
   return err
}

func (m *PTYManager) ReadFromSession(sessionID string) ([]byte, error) {
   m.mu.RLock()
   session, exists := m.sessions[sessionID]
   m.mu.RUnlock()

   if !exists {
      return nil, fmt.Errorf("session not found")
   }

   buf := make([]byte, 1024)
   n, err := session.Output.Read(buf)
   return buf[:n], err
}
```
