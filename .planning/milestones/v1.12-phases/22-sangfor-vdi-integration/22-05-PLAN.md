# Phase 22-05: VDI前端UI实现（完整VDI API集成）

## Objective

实现VDI虚拟机管理和服务器配置的前端用户界面，包括列表展示、完整操作按钮（创建、删除、开关机、重启、同步、配置IP、重命名、绑定用户）、虚拟机详情页面（含账号管理标签页）、配置表单和路由集成。所有操作都调用真实的VDI API。

**关键设计决策：账号管理功能集成在虚拟机详情页中**
- **避免并列菜单混淆**：不创建独立的"账号管理"二级菜单，因为账号属于虚拟机（数据模型关系：`vdi_vm_accounts.vm_id` 外键）
- **优化用户体验**：在虚拟机详情页的"账号管理"标签页中直接管理该VM的账号，避免菜单间跳转
- **遵循业界实践**：参考VMware vSphere、Azure Portal等成熟产品，都在虚拟机详情页管理该VM的用户/权限

**Purpose**: 提供友好的VDI管理界面，支持用户通过Web界面完成所有VDI管理操作

**Output**: 完整的前端VDI模块，所有虚拟机操作都调用VDI API，账号管理集成在虚拟机详情页

   ```typescript
   import { post } from './api';
   import type {
     BaseResponse,
     PageResponse,
   } from '@/types';
   import type {
     VirtualMachine,
     VDIServer,
     VMListParams,
     CreateVMRequest,
     UpdateVMRequest,
     VMOperateRequest,
     VMIPConfigRequest,
     RenameVMRequest,
     BindUserRequest,
     VDIServerConfig,
     VMPageResponse,
   } from '@/types/vdi';

   // ==================== 虚拟机 API（完整VDI API集成）====================

   export const vmApi = {
     // 基础CRUD
     list: async (params: VMListParams) => {
       return await post<VMPageResponse>('/vdi/vm/list', params);
     },

     get: async (id: string) => {
       return await post<BaseResponse<VirtualMachine>>(`/vdi/vm/${id}`, {});
     },

     create: async (data: CreateVMRequest) => {
       return await post<BaseResponse<VirtualMachine>>('/vdi/vm', data);
     },

     update: async (id: string, data: UpdateVMRequest) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/update`, data);
     },

     delete: async (id: string) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/delete`, {});
     },

     // VDI操作（调用VDI API）
     operate: async (request: VMOperateRequest) => {
       return await post<BaseResponse<void>>('/vdi/vm/operate', request);
     },

     configIP: async (configs: VMIPConfigRequest[]) => {
       return await post<BaseResponse<void>>('/vdi/vm/config_ip', configs);
     },

     rename: async (id: string, request: RenameVMRequest) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/rename`, request);
     },

     bindUser: async (id: string, request: BindUserRequest) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/bind_user`, request);
     },

     unbindUser: async (id: string) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/unbind_user`, {});
     },

     // 同步操作
     sync: async (id: string) => {
       return await post<BaseResponse<void>>(`/vdi/vm/${id}/sync`, {});
     },

     // 批量操作辅助方法
     batchOperate: async (vmIds: string[], action: string) => {
       return await post<BaseResponse<void>>('/vdi/vm/operate', {
         vm_ids: vmIds,
         action,
       });
     },
   };

   // ==================== VDI 服务器 API ====================

   export const vdiServerApi = {
     list: async (params: { current: number; pageSize: number }) => {
       return await post<PageResponse<VDIServer>>('/vdi/servers/list', params);
     },

     get: async (id: string) => {
       return await post<BaseResponse<VDIServer>>(`/vdi/servers/${id}`, {});
     },

     create: async (data: VDIServerConfig) => {
       return await post<BaseResponse<VDIServer>>('/vdi/servers', data);
     },

     update: async (id: string, data: Partial<VDIServerConfig>) => {
       return await post<BaseResponse<void>>(`/vdi/servers/${id}/update`, data);
     },

     delete: async (id: string) => {
       return await post<BaseResponse<void>>(`/vdi/servers/${id}/delete`, {});
     },

     testConnection: async (id: string) => {
       return await post<BaseResponse<void>>(`/vdi/servers/${id}/test`, {});
     },
   };

   // ==================== 类型导出 ====================

   export type {
     VirtualMachine,

     VMListParams,
     CreateVMRequest,
     UpdateVMRequest,
     VMOperateRequest,
     VMIPConfigRequest,
     RenameVMRequest,

     VDIServerConfig,
   };
   ```

     const [loading, setLoading] = useState(false);

     const [current, setCurrent] = useState(1);
     const [pageSize, setPageSize] = useState(10);
     const [selectedRowKeys, setSelectedRowKeys] = useState<React.Key[]>([]);
     const [filters, setFilters] = useState<Partial<VMListParams>>({});
     const [createModalVisible, setCreateModalVisible] = useState(false);
     const [configIPModalVisible, setConfigIPModalVisible] = useState(false);

     // 加载虚拟机列表
     const loadVMs = async () => {
       setLoading(true);
       try {
         const params: VMListParams = {
           current,
           pageSize,
           ...filters,
         };
         const result = await vmApi.list(params);
         setVMs(result.data.list || []);
         setTotal(result.data.total || 0);
       } catch (error) {
         message.error('加载虚拟机列表失败');
       } finally {
         setLoading(false);
       }
     };

       loadVMs();
     }, [current, pageSize, filters]);

       try {
         const values = await form.validateFields();

           ...values,

         await vmApi.create(data);
         message.success('虚拟机创建成功，VDI API调用完成');
         setCreateModalVisible(false);
         form.resetFields();
         loadVMs();
       } catch (error) {
         message.error('创建虚拟机失败');
       } finally {
         setLoading(false);
       }
     };

       const ids = vmIds || selectedRowKeys as string[];

         message.warning('请选择要操作的虚拟机');
         return;
       }

       setLoading(true);
       try {
         await vmApi.batchOperate(ids, action);
         message.success(`${action === 'start' ? '开机' : action === 'stop' ? '关机' : '重启'}操作已提交，VDI API调用成功`);
         setSelectedRowKeys([]);
         loadVMs();
       } catch (error) {
         message.error('操作失败，VDI API调用失败');
       } finally {
         setLoading(false);
       }
     };

     // 删除虚拟机（调用VDI API）
     const handleDelete = async (id: string) => {
       setLoading(true);
       try {
         await vmApi.delete(id);
         message.success('删除成功，VDI API调用完成');
         loadVMs();
       } catch (error) {
         message.error('删除失败，VDI API调用失败');
       } finally {
         setLoading(false);
       }

     // 同步虚拟机（调用VDI API）

       setLoading(true);
       try {
         await vmApi.sync(id);
         message.success('同步成功，VDI状态已更新');
         loadVMs();
       } catch (error) {
         message.error('同步失败');
       } finally {
         setLoading(false);
       }
     };

       try {
         const values = await form.validateFields();
         setLoading(true);
         await vmApi.configIP([{
           vm_id: selectedVM!.vm_id,
           ip_address: values.ip_address,
           netmask: values.netmask,

         }]);
         message.success('IP配置成功，VDI API调用完成');

       } catch (error) {

         setLoading(true);
         await vmApi.bindUser(selectedVM!.id, { user_id: values.user_id });
         message.success('用户绑定成功，VDI API调用完成');
