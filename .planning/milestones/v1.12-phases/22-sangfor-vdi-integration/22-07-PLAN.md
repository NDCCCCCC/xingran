   ```go
   func (c *agentClientImpl) getAgentEndpoint(ctx context.Context, agentID string) (string, error) {
       // 查询 Agent 信息获取 IP 地址
       var agent VDIAgent
       if err := c.db.WithContext(ctx).Where("agent_id = ? AND status = ?", agentID, "online").First(&agent).Error; err != nil {
           return "", fmt.Errorf("agent not found or offline: %w", err)
       }

       // 返回 Agent 的 HTTPS 端点
       return fmt.Sprintf("https://%s:8443", agent.IPAddress), nil
   }

   func (c *agentClientImpl) generateAgentToken(ctx context.Context, agentID string) (string, error) {
       // 为 Agent 生成 JWT 令牌
       claims := map[string]interface{}{
           "agent_id": agentID,
           "type":     "agent",
           "exp":      time.Now().Add(24 * time.Hour).Unix(),
       }

       return c.jwtManager.GenerateToken(claims)
   }

       endpoint, err := c.getAgentEndpoint(ctx, agentID)
       if err != nil {
           return err

       token, err := c.generateAgentToken(ctx, agentID)
       if err != nil {
           return err

       // 调用 Agent API
   ```

2. 实现账号服务：
   ```go
   package vdi

   import (
       "context"
       "fmt"
       "time"
       
       "github.com/google/uuid"
       "gorm.io/gorm"
       "github.com/xingran-next/xingran-go-backend/internal/models/operations"
   )

   type accountServiceImpl struct {
       db          *gorm.DB
       agentClient AgentClient
   }

   func NewAccountService(db *gorm.DB, agentClient AgentClient) AccountService {
       return &accountServiceImpl{
           db:          db,
           agentClient: agentClient,
       }
   }

   func (s *accountServiceImpl) CreateAccount(ctx context.Context, vmID string, req *CreateAccountRequest) (*VDIVMAccountDTO, error) {
       // 1. 查询 VM 信息
       var vm operations.VDIVirtualMachine
       if err := s.db.WithContext(ctx).Where("id = ?", vmID).First(&vm).Error; err != nil {
           return nil, fmt.Errorf("VM not found: %w", err)
       }

       // 2. 查询 VM 的 Agent
       var agent operations.VDIAgent
       if err := s.db.WithContext(ctx).Where("vm_id = ? AND status = ?", vmID, "online").First(&agent).Error; err != nil {
           return nil, fmt.Errorf("no online agent found for VM: %w", err)
       }

       // 3. 调用 Agent 创建账号
       agentReq := &CreateAccountRequest{
           Username: req.Username,
           Password: req.Password,
           OSType:   req.OSType,
           IsAdmin:  req.IsAdmin,
       }

       if err := s.agentClient.CreateAccount(ctx, agent.AgentID, agentReq); err != nil {
           // 记录审计日志（失败）
           s.recordAuditLog(ctx, vmID, "", "create", req.Username, err.Error(), "failed")
           return nil, fmt.Errorf("failed to create account in VM: %w", err)
       }

       // 4. 创建本地账号记录
       account := &operations.VDIVMAccount{
           VMID:       vmID,
           AccountID:  uuid.New().String(),
           Username:   req.Username,
           OSType:     req.OSType,
           IsAdmin:    req.IsAdmin,
           IsEnabled:  true,
           SyncStatus: "synced",
           VdiServerID: vm.VdiServerID,
       }

       if err := s.db.WithContext(ctx).Create(account).Error; err != nil {
           return nil, fmt.Errorf("failed to create account record: %w", err)
       }

       // 5. 记录审计日志（成功）
       s.recordAuditLog(ctx, vmID, account.AccountID, "create", req.Username, "", "success")

       return s.toAccountDTO(account), nil
   }

   func (s *accountServiceImpl) DeleteAccount(ctx context.Context, accountID string) error {
       // 1. 查询账号记录
       var account operations.VDIVMAccount
       if err := s.db.WithContext(ctx).Where("id = ?", accountID).First(&account).Error; err != nil {
           return fmt.Errorf("account not found: %w", err)
       }

       // 2. 查询 VM 的 Agent
       var agent operations.VDIAgent
       if err := s.db.WithContext(ctx).Where("vm_id = ? AND status = ?", account.VMID, "online").First(&agent).Error; err != nil {
           return fmt.Errorf("no online agent found for VM: %w", err)
       }

       // 3. 调用 Agent 删除账号
       if err := s.agentClient.DeleteAccount(ctx, agent.AgentID, account.Username); err != nil {
           s.recordAuditLog(ctx, account.VMID, account.AccountID, "delete", account.Username, err.Error(), "failed")
           return fmt.Errorf("failed to delete account in VM: %w", err)
       }

       // 4. 删除本地记录（软删除）
       if err := s.db.WithContext(ctx).Delete(&account).Error; err != nil {
           return fmt.Errorf("failed to delete account record: %w", err)
       }

       // 5. 记录审计日志
       s.recordAuditLog(ctx, account.VMID, account.AccountID, "delete", account.Username, "", "success")

       return nil
   }
   ```

   ```go
   // CreateAccount 创建账号
   // @Summary 创建 VM 账号
   // @Description 在指定虚拟机内创建本地账号
   // @Tags VDI账号管理
   // @Accept json
   // @Produce json
   // @Param vmId path string true "虚拟机ID"
   // @Param request body vdiServices.CreateAccountRequest true "账号信息"
   // @Success 200 {object} response.Response{data=vdiServices.VDIVMAccountDTO}
   // @Failure 400 {object} response.Response
   // @Router /vdi/vm/{vmId}/accounts [post]
   func (h *AccountHandler) CreateAccount(c *gin.Context) {

       if vmID == "" {
           response.Error(c, http.StatusBadRequest, "虚拟机ID不能为空")
           return
       }

       var req vdiServices.CreateAccountRequest
       if !handleJSONBinding(c, &req) {
           return
       }

       account, err := h.accountService.CreateAccount(c.Request.Context(), vmID, &req)
       if !handleServiceError(c, err, "创建账号") {
           return
       }

       response.Success(c, account)
   }

       vmID := c.Param("vmId")
   ```
