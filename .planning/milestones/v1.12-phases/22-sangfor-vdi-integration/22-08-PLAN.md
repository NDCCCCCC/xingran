# Phase 22-08: 密码轮换系统

## Objective

实现 VM 管理员密码自动轮换系统，支持定期自动修改密码、密码历史记录、密码策略配置和过期通知。这是 Phase 22 的核心安全功能。

**Purpose**: 解决 VM 管理员密码定期轮换的安全要求，确保密码定期更新且不中断服务

**Output**: 完整的密码轮换系统，支持自动轮换、策略配置和历史记录

       "context"

       "fmt"

       "strings"
       "time"

       db          *gorm.DB
       agentClient AgentClient
       config      *PasswordPolicy
   }

   func NewPasswordRotationService(db *gorm.DB, agentClient AgentClient) PasswordRotationService {
       return &passwordRotationServiceImpl{
           db:          db,
           agentClient: agentClient,
       }
   }

   // CheckAndRotate 检查并轮换到期的密码
   func (s *passwordRotationServiceImpl) CheckAndRotate(ctx context.Context) error {
       // 1. 加载密码策略
       policy, err := s.GetPasswordPolicy(ctx)
       if err != nil {
           return fmt.Errorf("failed to load password policy: %w", err)
       }

       if !policy.RotationEnabled {
           return nil // 密码轮换未启用
       }

       // 2. 查询需要轮换的账号
       accounts, err := s.getAccountsDueForRotation(ctx, policy)
       if err != nil {
           return fmt.Errorf("failed to query accounts: %w", err)
       }

       // 3. 为每个账号执行轮换
       results := make([]*RotationResult, 0, len(accounts))
       for _, account := range accounts {
           result, err := s.RotateAccountPassword(ctx, account.AccountID)
           if err != nil {
               // 记录错误但继续处理其他账号
               result = &RotationResult{
                   AccountID:    account.AccountID,
                   Success:      false,
                   ErrorMessage: err.Error(),
               }
           }
           results = append(results, result)
       }

       // 4. 发送通知（异步）
       go s.sendRotationSummaryNotifications(ctx, results)

       return nil
   }

   func (s *passwordRotationServiceImpl) getAccountsDueForRotation(ctx context.Context, policy *PasswordPolicy) ([]operations.VDIVMAccount, error) {
       var accounts []operations.VDIVMAccount
       
       // 查询条件：
       // 1. 账号已启用
       // 2. 密码即将到期（提前通知天数内）或已过期
       expiryThreshold := time.Now().Add(policy.AdvanceNotice * 24 * time.Hour)
       
       err := s.db.WithContext(ctx).
           Where("is_enabled = ?", true).
           Where("(password_expires_at IS NULL OR password_expires_at <= ?)", expiryThreshold).
           Find(&accounts).Error
           
       return accounts, err
   }

   func (s *passwordRotationServiceImpl) RotateAccountPassword(ctx context.Context, accountID string) (*RotationResult, error) {
       // 1. 查询账号信息
       var account operations.VDIVMAccount
       if err := s.db.WithContext(ctx).Where("account_id = ?", accountID).First(&account).Error; err != nil {
           return nil, fmt.Errorf("account not found: %w", err)
       }

       // 2. 生成安全随机密码
       newPassword, err := s.GenerateSecurePassword()
       if err != nil {
           return nil, fmt.Errorf("failed to generate password: %w", err)
       }

       // 3. 检查密码历史（防止重复）
       reuseAllowed, err := s.CheckPasswordReuse(ctx, accountID, newPassword)
       if err != nil {
           return nil, fmt.Errorf("failed to check password history: %w", err)
       }
       if !reuseAllowed {
           return nil, fmt.Errorf("password reuse detected")
       }

       // 4. 通过 Agent 调用 VM 本地命令修改密码
       // 注意：这里需要调用 Agent 的密码重置接口
       // if err := s.agentClient.ResetPassword(ctx, agentID, account.Username, newPassword); err != nil {
       //     return &RotationResult{
       //         AccountID:    accountID,
       //         Success:      false,
       //         ErrorMessage: err.Error(),
       //     }, nil
       // }

       // 5. 加密存储新密码
       encryptedPassword, err := s.encryptPassword(newPassword)
       if err != nil {
           return nil, fmt.Errorf("failed to encrypt password: %w", err)
       }

       // 6. 更新账号记录
       now := time.Now()
       expiryTime := now.Add(90 * 24 * time.Hour) // 90 天后过期
       
       updates := map[string]interface{}{
           "password_encrypted": encryptedPassword,
           "password_expires_at": &expiryTime,
           "password_changed_at": &now,
       }

       if err := s.db.WithContext(ctx).Model(&account).Updates(updates).Error; err != nil {
           return nil, fmt.Errorf("failed to update account: %w", err)
       }

       // 7. 记录密码历史
       oldPasswordHash := hashPassword("") // 旧密码哈希（如果可用）
       newPasswordHash := hashPassword(newPassword)
       
       historyReq := &PasswordHistoryRequest{
           AccountID:       accountID,
           OldPasswordHash: oldPasswordHash,
           NewPasswordHash: newPasswordHash,
           RotatedBy:       "system",
           Reason:          "Scheduled rotation",
       }
       
       if err := s.RecordPasswordHistory(ctx, historyReq); err != nil {
           // 非关键错误，记录日志但继续
       }

       // 8. 返回成功结果
       return &RotationResult{
           AccountID:   accountID,
           Success:     true,
           RotatedAt:   now,
           NewPassword: newPassword, // 仅用于测试，生产环境不应返回
       }, nil
   }

   func (s *passwordRotationServiceImpl) GenerateSecurePassword() (string, error) {
       policy, err := s.GetPasswordPolicy(context.Background())
       if err != nil {
           policy = &PasswordPolicy{
               MinLength:        16,
               RequireUppercase: true,
               RequireLowercase: true,
               RequireNumber:    true,
               RequireSpecial:   true,
           }
       }

       var password strings.Builder
       charsets := []string{}

       if policy.RequireUppercase {
           charsets = append(charsets, "ABCDEFGHIJKLMNOPQRSTUVWXYZ")
       }
       if policy.RequireLowercase {
           charsets = append(charsets, "abcdefghijklmnopqrstuvwxyz")
       }
       if policy.RequireNumber {
           charsets = append(charsets, "0123456789")
       }
       if policy.RequireSpecial {
           charsets = append(charsets, "!@#$%^&*()_+-=[]{}|;:,.<>?")
       }

       // 确保每个字符集至少有一个字符
       for _, charset := range charsets {
           n, err := rand.Int(rand.Reader, big.NewInt(int64(len(charset))))
           if err != nil {
               return "", err
           }
           password.WriteString(charset[n.Int64()])
       }

       // 填充剩余长度
       allCharsets := strings.Join(charsets, "")
       for password.Len() < policy.MinLength {
           n, err := rand.Int(rand.Reader, big.NewInt(int64(len(allCharsets))))
           if err != nil {
               return "", err
           }
           password.WriteString(allCharsets[n.Int64()])
       }

       // 打乱字符顺序
       return shuffleString(password.String()), nil
   }

   func (s *passwordRotationServiceImpl) ValidatePassword(password string) error {
       policy, err := s.GetPasswordPolicy(context.Background())
       if err != nil {
           return err
       }

       if len(password) < policy.MinLength {
           return fmt.Errorf("password too short (min %d characters)", policy.MinLength)
       }

       if policy.RequireUppercase && !strings.ContainsAny(password, "ABCDEFGHIJKLMNOPQRSTUVWXYZ") {
           return fmt.Errorf("password must contain uppercase letters")
       }

       if policy.RequireLowercase && !strings.ContainsAny(password, "abcdefghijklmnopqrstuvwxyz") {
           return fmt.Errorf("password must contain lowercase letters")
       }

       if policy.RequireNumber && !strings.ContainsAny(password, "0123456789") {
           return fmt.Errorf("password must contain numbers")
       }

       if policy.RequireSpecial && !strings.ContainsAny(password, "!@#$%^&*()_+-=[]{}|;:,.<>?") {
           return fmt.Errorf("password must contain special characters")
       }

       return nil
   }

   func (s *passwordRotationServiceImpl) RecordPasswordHistory(ctx context.Context, req *PasswordHistoryRequest) error {
       history := &operations.VDIPasswordHistory{
           AccountID:       req.AccountID,
           OldPasswordHash: req.OldPasswordHash,
           NewPasswordHash: req.NewPasswordHash,
           RotatedAt:       time.Now(),
           RotatedBy:       req.RotatedBy,
           Reason:          req.Reason,
       }

       return s.db.WithContext(ctx).Create(history).Error
   }

   func (s *passwordRotationServiceImpl) CheckPasswordReuse(ctx context.Context, accountID, newPassword string) (bool, error) {
       policy, err := s.GetPasswordPolicy(ctx)
       if err != nil {
           return false, err
       }

       newPasswordHash := hashPassword(newPassword)

       // 查询最近 N 次的密码历史
       var history []operations.VDIPasswordHistory
       err = s.db.WithContext(ctx).
           Where("account_id = ?", accountID).
           Order("rotated_at DESC").
           Limit(policy.HistoryCount).
           Find(&history).Error

       if err != nil {
           return false, err
       }

       // 检查新密码是否在历史中
       for _, h := range history {
           if h.NewPasswordHash == newPasswordHash {
           return false, nil // 密码重复，不允许使用
       }
       }

       return true, nil // 密码未重复，可以使用
   }

   func (s *passwordRotationServiceImpl) GetPasswordPolicy(ctx context.Context) (*PasswordPolicy, error) {
       // 从 sys_config 表加载密码策略
       configs := make(map[string]string)
       rows, err := s.db.WithContext(ctx).
           Table("sys_config").
           Where("config_key LIKE ?", "vdi.password.%").
           Rows()

       if err != nil {
           return nil, err
       }
       defer rows.Close()

       for rows.Next() {
           var key, value string
           if err := rows.Scan(&key, &value); err != nil {
               continue
           }
           configs[key] = value
       }

       // 解析配置
       policy := &PasswordPolicy{
           RotationEnabled:  configs["vdi.password.rotation_enabled"] == "true",
           AdvanceNotice:     parseInt(configs["vdi.password.rotation_advance"], 7),
           MinLength:         parseInt(configs["vdi.password.min_length"], 16),
           RequireUppercase:  configs["vdi.password.require_uppercase"] == "true",
           RequireLowercase:  configs["vdi.password.require_lowercase"] == "true",
           RequireNumber:     configs["vdi.password.require_number"] == "true",
           RequireSpecial:    configs["vdi.password.require_special"] == "true",
           HistoryCount:      parseInt(configs["vdi.password.history_count"], 5),
       }

       if interval := parseInt(configs["vdi.password.rotation_interval"], 90); interval > 0 {
           policy.RotationInterval = time.Duration(interval) * 24 * time.Hour
       }

       return policy, nil
   }

   func (s *passwordRotationServiceImpl) UpdatePasswordPolicy(ctx context.Context, policy *PasswordPolicy) error {

   func hashPassword(password string) string {
       hash, _ := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
       return string(hash)
   }

   func shuffleString(s string) string {
       // Fisher-Yates 洗牌算法
       runes := []rune(s)
       n := len(runes)
       for i := n - 1; i > 0; i-- {
           j, _ := rand.Int(rand.Reader, big.NewInt(int64(i + 1)))
           runes[i], runes[j.Int64()] = runes[j.Int64()], runes[i]
       }
       return string(runes)
   }

   func parseInt(s string, defaultValue int) int {
       if s == "" {
           return defaultValue
       }
       var result int
       fmt.Sscanf(s, "%d", &result)
       return result
   }

### Task 4: 创建密码策略配置 API

**Files**: `internal/api/v1/vdi/password_policy_handler.go`, `internal/api/v1/vdi/password_policy_router.go`

**Action**:
1. 创建密码策略处理器：
   ```go
   package vdi

   import (
       "net/http"
       
       "github.com/gin-gonic/gin"
       vdiServices "github.com/xingran-next/xingran-go-backend/internal/services/vdi"
       "github.com/xingran-next/xingran-go-backend/pkg/response"
   )

   type PasswordPolicyHandler struct {
       passwordService vdiServices.PasswordRotationService
   }

   func NewPasswordPolicyHandler(passwordService vdiServices.PasswordRotationService) *PasswordPolicyHandler {
       return &PasswordPolicyHandler{passwordService: passwordService}
   }

   // GetPasswordPolicy 获取密码策略
   func (h *PasswordPolicyHandler) GetPasswordPolicy(c *gin.Context) {
       policy, err := h.passwordService.GetPasswordPolicy(c.Request.Context())
       if err != nil {
           response.Error(c, http.StatusInternalServerError, err.Error())
           return
       }

       response.Success(c, policy)
   }

   // UpdatePasswordPolicy 更新密码策略
   func (h *PasswordPolicyHandler) UpdatePasswordPolicy(c *gin.Context) {
       var policy vdiServices.PasswordPolicy
       if err := c.ShouldBindJSON(&policy); err != nil {
           response.Error(c, http.StatusBadRequest, "请求参数错误")
           return
       }

       if err := h.passwordService.UpdatePasswordPolicy(c.Request.Context(), &policy); err != nil {
           response.Error(c, http.StatusInternalServerError, err.Error())
           return
       }

       response.Success(c, nil)
   }

   // RotatePassword 手动触发密码轮换
   func (h *PasswordPolicyHandler) RotatePassword(c *gin.Context) {
       var req struct {
           AccountID string `json:"account_id" validate:"required"`
       }

       if err := c.ShouldBindJSON(&req); err != nil {
           response.Error(c, http.StatusBadRequest, "请求参数错误")
           return
       }

       result, err := h.passwordService.RotateAccountPassword(c.Request.Context(), req.AccountID)
       if err != nil {
           response.Error(c, http.StatusInternalServerError, err.Error())
           return
       }

       response.Success(c, result)
   }
   ```

2. 创建路由配置：
   ```go
   package vdi

   import (
       "github.com/gin-gonic/gin"
       "github.com/xingran-next/xingran-go-backend/internal/core"
       vdiServices "github.com/xingran-next/xingran-go-backend/internal/services/vdi"
   )

   func SetupPasswordPolicyRouter(r *gin.RouterGroup, core *core.Core) {
       passwordService := vdiServices.NewPasswordRotationService(core.GetDB(), nil)
       handler := NewPasswordPolicyHandler(passwordService)

       r.GET("/policy", handler.GetPasswordPolicy)
       r.POST("/policy", handler.UpdatePasswordPolicy)
       r.POST("/rotate", handler.RotatePassword)
   }
   ```
