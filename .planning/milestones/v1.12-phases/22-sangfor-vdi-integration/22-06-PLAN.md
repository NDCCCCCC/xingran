# Phase 22-06: VM Agent 服务实现

## Objective

实现部署在虚拟机内的 Agent 服务，支持账号管理、Web Console 终端连接和心跳上报。Agent 使用 JWT 令牌向后端认证，独立于 VM 管理员密码工作。

**Purpose**: 解决 VM 管理员密码定期轮换导致的连接失效问题，提供可靠的账号管理和终端访问能力

**Output**: 可部署的跨平台 Agent 服务（Windows + Linux）

### Agent 架构决策（来自 CONTEXT.md）

**D-01: VM连接方式架构决策 → 选择 VM Agent 架构**
- SSH/WinRM 方案需要存储 VM 管理员密码 → 定期修改导致失效
- Agent 方案使用 JWT 令牌，完全解耦对 VM 管理员密码的依赖

**D-02: Agent功能范围**
- ✅ 账号 CRUD 操作（创建、删除、启用、禁用）
- ✅ 密码重置功能
- ✅ 系统状态监控（定期上报）
- ✅ Web Console（网页终端）
- ❌ 远程命令执行（风险过高）
- ❌ 文件传输（Web Console 仅用于终端）

**D-03/D-04: Agent 权限设计**
- Windows: 受限管理员账号（Restricted Management）
- Linux: 专用服务账号 + 受限 sudo

**D-05: Agent 安全机制**
- JWT 令牌认证（24 小时有效期，自动刷新）
- TLS 1.3 加密通信
- 操作审计日志
- 速率限制

   ```go
       return nil

   // ValidateToken 验证令牌有效性
   func (a *JWTAuthenticator) ValidateToken(tokenString string) (*jwt.Claims, error) {
       token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {
           if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
               return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
           }
           return []byte(a.secret), nil
       })

       if err != nil {
           return nil, err
       }

       if claims, ok := token.Claims.(jwt.MapClaims); ok && token.Valid {
           return &claims, nil
       }

       return nil, fmt.Errorf("invalid token")
   }

   // GetCurrentToken 获取当前有效令牌
   func (a *JWTAuthenticator) GetCurrentToken(ctx context.Context) (string, error) {
       if err := a.RefreshToken(ctx); err != nil {
           return "", err
       }
       return a.currentToken, nil
   }
   ```

### Task 3: 实现账号管理器

1. 创建账号管理器（跨平台抽象）：
   ```go
   package server

   import (
       "context"
       "fmt"
       "os/exec"
       "runtime"
   )

   type AccountManager struct {
       platform string
   }

   func NewAccountManager() *AccountManager {
       return &AccountManager{
           platform: runtime.GOOS,
       }
   }

   type Account struct {
       Username   string `json:"username"`
       Password   string `json:"password,omitempty"`
       IsAdmin    bool   `json:"is_admin"`
       IsEnabled  bool   `json:"is_enabled"`
       OSType     string `json:"os_type"`
   }

   // CreateAccount 创建本地账号
   func (m *AccountManager) CreateAccount(ctx context.Context, account *Account) error {
       switch m.platform {
       case "windows":
           return m.createWindowsAccount(ctx, account)
       case "linux":
           return m.createLinuxAccount(ctx, account)
       default:
           return fmt.Errorf("unsupported platform: %s", m.platform)
       }
   }

   // createWindowsAccount 创建 Windows 账号
   func (m *AccountManager) createWindowsAccount(ctx context.Context, account *Account) error {
       // 使用 PowerShell 创建账号
       cmd := exec.CommandContext(ctx, "powershell", "-Command",
           fmt.Sprintf("New-LocalUser -Name %s -Password (ConvertTo-SecureString %s -AsPlainText -Force) -Description 'XingRan VDI Account'",
               account.Username, account.Password))
       
       if account.IsAdmin {
           // 添加到管理员组
           cmd2 := exec.CommandContext(ctx, "powershell", "-Command",
               fmt.Sprintf("Add-LocalGroupMember -Group 'Administrators' -Member %s", account.Username))
           if err := cmd2.Run(); err != nil {
               return fmt.Errorf("failed to add to admin group: %w", err)
           }
       }

       return cmd.Run()
   }

   // createLinuxAccount 创建 Linux 账号
   func (m *AccountManager) createLinuxAccount(ctx context.Context, account *Account) error {
       // 创建用户
       cmd := exec.CommandContext(ctx, "useradd", "-m", account.Username)
       if err := cmd.Run(); err != nil {
           return fmt.Errorf("failed to create user: %w", err)
       }

       // 设置密码
       cmd2 := exec.CommandContext(ctx, "chpasswd")
       stdin, err := cmd2.StdinPipe()
       if err != nil {
           return err
   }
       go func() {
       defer stdin.Close()
       fmt.Fprintf(stdin, "%s:%s\n", account.Username, account.Password)
   }()
       if err := cmd2.Run(); err != nil {
       return fmt.Errorf("failed to set password: %w", err)
   }

   if account.IsAdmin {
       // 配置 sudo 权限
       return m.configureSudo(ctx, account.Username)
   }

   return nil
   }

   // DeleteAccount 删除账号
   func (m *AccountManager) DeleteAccount(ctx context.Context, username string) error {
       switch m.platform {
       case "windows":
           return exec.CommandContext(ctx, "powershell", "-Command",
               fmt.Sprintf("Remove-LocalUser -Name %s", username)).Run()
       case "linux":
           return exec.CommandContext(ctx, "userdel", "-r", username).Run()
       default:
           return fmt.Errorf("unsupported platform: %s", m.platform)
       }
   }

   // ResetPassword 重置密码
   func (m *AccountManager) ResetPassword(ctx context.Context, username, newPassword string) error {
       switch m.platform {
       case "windows":
           return exec.CommandContext(ctx, "powershell", "-Command",
               fmt.Sprintf("Set-LocalUser -Name %s -Password (ConvertTo-SecureString %s -AsPlainText -Force)",
                   username, newPassword)).Run()
       case "linux":
           cmd := exec.CommandContext(ctx, "chpasswd")
           stdin, err := cmd.StdinPipe()
           if err != nil {
               return err
           }
           go func() {
               defer stdin.Close()
               fmt.Fprintf(stdin, "%s:%s\n", username, newPassword)
           }()
           return cmd.Run()
       default:
           return fmt.Errorf("unsupported platform: %s", m.platform)
       }
   }

   // EnableAccount 启用账号
   func (m *AccountManager) EnableAccount(ctx context.Context, username string) error {
       switch m.platform {
       case "windows":
           return exec.CommandContext(ctx, "powershell", "-Command",
               fmt.Sprintf("Enable-LocalUser -Name %s", username)).Run()
       case "linux":
           return exec.CommandContext(ctx, "usermod", "-U", username).Run()
       default:
           return fmt.Errorf("unsupported platform: %s", m.platform)
       }
   }

   // DisableAccount 禁用账号
   func (m *AccountManager) DisableAccount(ctx context.Context, username string) error {
       switch m.platform {
       case "windows":
           return exec.CommandContext(ctx, "powershell", "-Command",
               fmt.Sprintf("Disable-LocalUser -Name %s", username)).Run()
       case "linux":
           return exec.CommandContext(ctx, "usermod", "-L", username).Run()
       default:
           return fmt.Errorf("unsupported platform: %s", m.platform)
       }
   }

   func (m *AccountManager) configureSudo(ctx context.Context, username string) error {
       // 写入 /etc/sudoers.d/xingran-agent
       sudoers := fmt.Sprintf("%s ALL=(root) NOPASSWD: /usr/sbin/useradd, /usr/sbin/userdel, /usr/sbin/usermod, /usr/bin/passwd\n", username)
       cmd := exec.CommandContext(ctx, "tee", fmt.Sprintf("/etc/sudoers.d/%s", username))
       cmd.Stdin = strings.NewReader(sudoers)
       if err := cmd.Run(); err != nil {
           return fmt.Errorf("failed to configure sudo: %w", err)
       }

       // 设置权限
       return exec.CommandContext(ctx, "chmod", "440", fmt.Sprintf("/etc/sudoers.d/%s", username)).Run()
   }
   ```

   ```go
   package server

   import (
       "net/http"
       "strings"
       
       "github.com/gin-gonic/gin"
   )

   func JWTAuth(authenticator *JWTAuthenticator) gin.HandlerFunc {
       return func(c *gin.Context) {
           authHeader := c.GetHeader("Authorization")
           if authHeader == "" {
               c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "missing authorization header"})
               return
           }

           tokenString := strings.TrimPrefix(authHeader, "Bearer ")
           claims, err := authenticator.ValidateToken(tokenString)
           if err != nil {
               c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "invalid token"})
               return
           }

           c.Set("agent_id", claims["agent_id"])
           c.Set("vm_id", claims["vm_id"])
           c.Next()
       }
   }
   ```

   ```go
   package server

   import (
       "net/http"
       
       "github.com/gin-gonic/gin"
   )

   type AgentHandler struct {
       accountManager *AccountManager
       authenticator  *JWTAuthenticator
   }

   func NewAgentHandler(accountManager *AccountManager, authenticator *JWTAuthenticator) *AgentHandler {
       return &AgentHandler{
           accountManager: accountManager,
           authenticator:  authenticator,
       }
   }

   // RegisterRoutes 注册路由
   func (h *AgentHandler) RegisterRoutes(r *gin.Engine) {
       api := r.Group("/api/v1")
       
       // 公开端点
       api.POST("/health", h.HealthCheck)
       api.POST("/register", h.Register)
       
       // 需要 JWT 认证的端点
       auth := api.Group("")
       auth.Use(JWTAuth(h.authenticator))
       {
           auth.POST("/accounts", h.CreateAccount)
           auth.DELETE("/accounts/:username", h.DeleteAccount)
           auth.POST("/accounts/:username/reset", h.ResetPassword)
           auth.POST("/accounts/:username/enable", h.EnableAccount)
           auth.POST("/accounts/:username/disable", h.DisableAccount)
           auth.GET("/console/ws", h.WebSocketTerminal)
           auth.POST("/heartbeat", h.Heartbeat)
       }
   }

   // CreateAccount 创建账号
   func (h *AgentHandler) CreateAccount(c *gin.Context) {
       var req Account
       if err := c.ShouldBindJSON(&req); err != nil {
           c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
           return
       }

       if err := h.accountManager.CreateAccount(c.Request.Context(), &req); err != nil {
           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Account created successfully"})
   }

   // DeleteAccount 删除账号
   func (h *AgentHandler) DeleteAccount(c *gin.Context) {
       username := c.Param("username")
       
       if err := h.accountManager.DeleteAccount(c.Request.Context(), username); err != nil {
           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Account deleted successfully"})
   }

   // ResetPassword 重置密码
   func (h *AgentHandler) ResetPassword(c *gin.Context) {
       username := c.Param("username")
       var req struct {
           NewPassword string `json:"new_password"`
       }
       
       if err := c.ShouldBindJSON(&req); err != nil {
           c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
           return
       }

       if err := h.accountManager.ResetPassword(c.Request.Context(), username, req.NewPassword); err != nil {
           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Password reset successfully"})
   }

   // EnableAccount 启用账号
   func (h *AgentHandler) EnableAccount(c *gin.Context) {
       username := c.Param("username")
       
       if err := h.accountManager.EnableAccount(c.Request.Context(), username); err != nil {
           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Account enabled successfully"})
   }

   // DisableAccount 禁用账号
   func (h *AgentHandler) DisableAccount(c *gin.Context) {
       username := c.Param("username")
       
       if err := h.accountManager.DisableAccount(c.Request.Context(), username); err != nil {
           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Account disabled successfully"})
   }

   // HealthCheck 健康检查
   func (h *AgentHandler) HealthCheck(c *gin.Context) {
       c.JSON(http.StatusOK, gin.H{"status": "healthy"})
   }

   // Register Agent 注册

           c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
           return
       }

       c.JSON(http.StatusOK, gin.H{"message": "Agent registered successfully"})
   ```

1. 创建 Agent 入口：

   ```go
   package main

   import (
       "context"
       "log"
       "os"
       "os/signal"
       "syscall"
       
       "github.com/gin-gonic/gin"
       "github.com/xingran-next/xingran-go-backend/internal/agent/server"
   )

   func main() {
       // 加载配置
       config, err := server.LoadConfig("/etc/xingran-agent/config.yaml")
       if err != nil {
           log.Fatalf("Failed to load config: %v", err)
       }

       // 初始化组件
       authenticator := server.NewJWTAuthenticator(config.JWTSecret, config.BackendURL)
       accountManager := server.NewAccountManager()
       handler := server.NewAgentHandler(accountManager, authenticator)

       // 注册 Agent
       if err := authenticator.Register(context.Background(), config.AgentID, config.VMID); err != nil {
           log.Fatalf("Failed to register agent: %v", err)
       }

       // 启动 HTTP 服务器
       r := gin.Default()
       handler.RegisterRoutes(r)

       go func() {
           if config.TLSCertFile != "" {
               log.Printf("Starting HTTPS server on %s", config.ListenAddr)
               if err := r.RunTLS(config.ListenAddr, config.TLSCertFile, config.TLSKeyFile); err != nil {
                   log.Fatalf("Failed to start server: %v", err)
               }
           } else {
               log.Printf("Starting HTTP server on %s", config.ListenAddr)
               if err := r.Run(config.ListenAddr); err != nil {
                   log.Fatalf("Failed to start server: %v", err)
               }
           }
       }()

       // 等待信号
       quit := make(chan os.Signal, 1)
       signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
       <-quit

       log.Println("Agent shutting down...")
   }
   ```

   ```bash
   WorkingDirectory=$AGENT_DIR
   ExecStart=$AGENT_BIN --config=$AGENT_DIR/config.yaml
   Restart=always
   RestartSec=10

   [Install]
   WantedBy=multi-user.target
   EOF

   systemctl enable xingran-vm-agent
   systemctl start xingran-vm-agent
   ```
