### API认证流程

```
POST /v1/auth/tokens
{
  "auth": {
    "name": "admin",
    "password": "password"
  }
}

响应:
{
  "error_code": 0,
  "error_message": "",
  "data": {
    "token": {
      "tenant_id": 0,
      "auth_token": "eyJ0eXAiOiJKV1QiLCJhbGc..."
    }
  }
}

后续请求需要在HTTP头中携带：
Auth-Token: {auth_token}
```

### Key Design Patterns

1. **接口隔离**: VDIClient接口定义明确，与实现分离
2. **错误包装**: 使用`fmt.Errorf`包装HTTP错误，保留调用栈
3. **Token缓存**: AuthToken缓存在数据库VDIServer表中，避免重复认证
4. **自动刷新**: Token过期前自动刷新（提前5分钟）
5. **重试机制**: 网络错误自动重试3次，指数退避
6. **密码安全**: 使用AD域模块的decryptVDIPassword解密密码

### Task 2: 实现认证和token管理

2. 实现认证逻辑：
   ```go
   package vdi

   import (
       "context"
       "fmt"
       "time"
       "github.com/xingran-next/xingran-go-backend/internal/models/operations"
       "gorm.io/gorm"
   )

   type VDIAuthManager struct {
       db         *gorm.DB
       httpClient *http.Client
       config     config.VDIServerConfig
       serverID   string // VDI服务器ID
   }

   func NewVDIAuthManager(db *gorm.DB, serverID string, cfg config.VDIServerConfig) *VDIAuthManager {
       return &VDIAuthManager{
           db:         db,
           httpClient: &http.Client{Timeout: 30 * time.Second},
           config:     cfg,
           serverID:   serverID,
       }
   }

   func (a *VDIAuthManager) Authenticate(ctx context.Context) (string, error) {
       // 1. 检查缓存的token是否有效
       var server operations.VDIServer
       err := a.db.WithContext(ctx).Where("id = ?", a.serverID).First(&server).Error
       if err == nil && server.AuthToken != "" && !a.isTokenExpired(server.TokenExpiry) {
           return server.AuthToken, nil
       }

       // 2. 从数据库解密密码
       password := operations.DecryptVDIPassword(server.PasswordEncrypted)
       if password == "" {
           return "", fmt.Errorf("failed to decrypt VDI server password")
       }

       // 3. 调用VDI API认证
       req := AuthRequest{
           Auth: AuthCredentials{
               Name:     a.config.Username,
               Password: password, // 使用解密后的密码
           },
       }

       var resp AuthResponse
       if err := a.callAPI(ctx, "/v1/auth/tokens", req, &resp); err != nil {
           return "", fmt.Errorf("authentication failed: %w", err)
       }

       if resp.ErrorCode != 0 {
           return "", &VDIError{Code: resp.ErrorCode, Message: resp.ErrorMessage}
       }

       // 4. 缓存token到数据库
       token := resp.Data.Token.AuthToken
       expiry := time.Now().Add(23 * time.Hour) // 提前1小时过期

       if err := a.cacheToken(ctx, token, expiry); err != nil {
           return "", fmt.Errorf("failed to cache token: %w", err)
       }

       return token, nil
   }

   func (a *VDIAuthManager) isTokenExpired(expiry *time.Time) bool {
       if expiry == nil {
           return true
       }
       return time.Now().After(*expiry)
   }

   func (a *VDIAuthManager) cacheToken(ctx context.Context, token string, expiry time.Time) error {
       return a.db.WithContext(ctx).
           Model(&operations.VDIServer{}).
           Where("id = ?", a.serverID).
           Updates(map[string]interface{}{
               "auth_token":    token,
               "token_expiry":  &expiry,
           }).Error
   }

   func (a *VDIAuthManager) callAPI(ctx context.Context, path string, body, result interface{}) error {
       url := fmt.Sprintf("%s%s", a.config.Endpoint, path)

       jsonBody, err := json.Marshal(body)
       if err != nil {
           return err
       }

       req, err := http.NewRequestWithContext(ctx, "POST", url, bytes.NewBuffer(jsonBody))
       if err != nil {
           return err
       }

       req.Header.Set("Content-Type", "application/json")

       // 重试逻辑
       var lastErr error
       for i := 0; i < 3; i++ {
           resp, err := a.httpClient.Do(req)
           if err != nil {
               lastErr = err
               time.Sleep(time.Second * time.Duration(i+1)) // 指数退避
               continue
           }
           defer resp.Body.Close()

           if resp.StatusCode != http.StatusOK {
               return fmt.Errorf("unexpected status code: %d", resp.StatusCode)
           }

           return json.NewDecoder(resp.Body).Decode(result)
       }

       return fmt.Errorf("HTTP request failed after retries: %w", lastErr)
   }
   ```

   func (c *vdiClientImpl) callAPI(ctx context.Context, token, method, path string, body, result interface{}) error {
       url := fmt.Sprintf("%s%s", c.config.Endpoint, path)

       var req *http.Request
       var err error

       if body != nil {
           jsonBody, err := json.Marshal(body)
           if err != nil {
               return err
           }
           req, err = http.NewRequestWithContext(ctx, method, url, bytes.NewBuffer(jsonBody))
       } else {
           req, err = http.NewRequestWithContext(ctx, method, url, nil)
       }

       if err != nil {
           return err
       }

       req.Header.Set("Auth-Token", token)
       req.Header.Set("Content-Type", "application/json")

       // 重试逻辑
       var lastErr error
       for i := 0; i < 3; i++ {
           resp, err := c.httpClient.Do(req)
           if err != nil {
               lastErr = err
               time.Sleep(time.Second * time.Duration(i+1))
               continue
           }
           defer resp.Body.Close()

           if resp.StatusCode != http.StatusOK {
               return fmt.Errorf("unexpected status code: %d", resp.StatusCode)
           }

           return json.NewDecoder(resp.Body).Decode(result)
       }

       return fmt.Errorf("HTTP request failed after retries: %w", lastErr)
   }
