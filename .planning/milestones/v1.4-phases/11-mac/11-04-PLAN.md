# Test Pattern (from existing portcollection tests)

Reference for test structure:
```go
package portcollection

import (
    "testing"
    "github.com/stretchr/testify/assert"
)

func TestParsePortStatus(t *testing.T) {
    tests := []struct {
        name     string
        input    string
        expected PortStatus
        wantErr  bool
    }{
        {
            name:  "Valid port up",
            input: "GigabitEthernet0/0/1 is up",
            expected: PortStatus{
                Interface: "GigabitEthernet0/0/1",
                Status:    "up",
            },
            wantErr: false,
        },
        // ... more test cases
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result, err := ParsePortStatus(tt.input)
            if tt.wantErr {
                assert.Error(t, err)
            } else {
                assert.NoError(t, err)
                assert.Equal(t, tt.expected, result)
            }
        })
    }
}
```
