# Priority-Based Rule Resolution (from RESEARCH.md example 3)

```go
func (s *FilterRuleService) GetEffectiveRule(device *models.NetworkDevice) (*MACFilterRule, error) {
    var rule MACFilterRule

    // Find most specific rule (vendor + device type)
    err := s.db.Where("device_type = ? AND vendor = ?", device.DeviceType, device.Vendor).
        Order("priority DESC").
        First(&rule).Error

    if err == nil {
        return &rule, nil
    }

    // Fallback to device type only
    err = s.db.Where("device_type = ? AND (vendor = '' OR vendor IS NULL)", device.DeviceType).
        Order("priority DESC").
        First(&rule).Error

    if err == nil {
        return &rule, nil
    }

    // Return default rule
    return &MACFilterRule{
        MACThreshold:     10,
        EnableLLDPFilter: true,
        Priority:         0,
    }, nil
}
```
