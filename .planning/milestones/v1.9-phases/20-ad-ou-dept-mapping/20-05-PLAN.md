func TestCountTotalDepts(t *testing.T) {
    db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{})
    assert.NoError(t, err)

    service := &DeptToADSyncService{db: db}

    // 测试空部门列表
    depts := []*Department{}
    count := service.countTotalDepts(depts)
    assert.Equal(t, 0, count)

    // 测试单层部门
    depts = []*Department{
        {DeptName: "Dept1"},
        {DeptName: "Dept2"},
    }
    count = service.countTotalDepts(depts)
    assert.Equal(t, 2, count)

    // 测试多层部门
    parent := &Department{DeptName: "Parent"}
    child1 := &Department{DeptName: "Child1"}
    child2 := &Department{DeptName: "Child2"}
    grandchild := &Department{DeptName: "Grandchild"}

    parent.Children = []*Department{child1, child2}
    child1.Children = []*Department{grandchild}

    depts = []*Department{parent}
    count = service.countTotalDepts(depts)
    assert.Equal(t, 4, count) // Parent + Child1 + Child2 + Grandchild
}
