    } from './sm4';

    describe('SM4 Encryption Utils', () => {
      it('should generate valid SM4 key', () => {
        const key = generateSM4Key();
        expect(key).toHaveLength(64); // 32字节 = 64个十六进制字符
        expect(/^[0-9a-f]{64}$/.test(key)).toBe(true);
      });

      it('should encrypt and decrypt correctly', async () => {
        const plaintext = 'Hello, SM4!';
        const key = generateSM4Key();
        const iv = generateIV();

        const cipher = await encryptSM4CBC(plaintext, key, iv);
        const decrypted = await decryptSM4CBC(cipher, key, iv);

        expect(decrypted).toBe(plaintext);
      });

      it('should generate different IV each time', () => {
        const iv1 = generateIV();
        const iv2 = generateIV();
        expect(iv1).not.toBe(iv2);
      });

      it('should handle empty input', async () => {
        const result = await encryptSM4CBC('', generateSM4Key(), generateIV());
        expect(result).toBe('');
      });
    });
