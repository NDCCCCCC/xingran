       ```typescript
       import { sm2 } from './sm2'; // 导入SM2工具

       /**
        * 加密登录密码（SM4-ECB模式）
        * 用于AD域控登录时的密码字段加密
        *
        * @param password 明文密码
        * @param key SM4密钥（16字节，32个十六进制字符）
        * @returns 加密后的密文（十六进制字符串）
        */
       export async function encryptPasswordWithSM4(password: string, key: string): Promise<string> {
         if (!password) {
           return '';
         }

         const sm4 = await getSM4();

         // 使用SM4-ECB模式加密密码（更简单，无需IV）
         const cipherHex = sm4.encrypt(password, key, { mode: 'ecb' });

         return cipherHex;
       }

       /**
        * 从后端获取SM4密钥
        * 密钥通过SM2公钥加密传输，保证密钥安全
        *
        * @returns SM4密钥（32个十六进制字符）
        */
       export async function fetchSM4KeyForPassword(): Promise<string> {
         try {
           // 调用后端API获取加密的SM4密钥
           const response = await fetch('/api/v1/system/auth/sm4-key');
           const result = await response.json();

           if (result.code === 0 && result.data?.encryptedKey) {
             // 使用SM2私钥解密密钥
             const sm4KeyHex = sm2.decrypt(result.data.encryptedKey);
             return sm4KeyHex;
           }

           throw new Error('获取SM4密钥失败');
         } catch (error) {
           console.error('[SM4] 获取密钥失败:', error);
           throw error;
         }
       }

       /**
        * 生成会话密钥（SM4密钥）
        * 用于当前登录会话的密码加密
        *
        * @returns 32字符的十六进制SM4密钥
        */
       export function generateSessionKey(): string {
         return generateSM4Key();
       }
     ```

    关键设计点：
    - 使用SM4-ECB模式（比CBC更简单，无需IV）
    - 支持从后端获取SM4密钥（SM2加密传输）
    - 提供会话密钥生成函数
    - 与现有SM2工具集成

       }

       export const useAuthStore = create<AuthState>((set, get) => ({
         // ... 现有状态

         login: async (loginData: LoginRequest) => {
           const { authMode = 'local' } = loginData; // 获取认证模式

           try {
             // 调用后端登录接口
             const response = await post('/system/auth/login', {
               username: loginData.username,
               password: loginData.password,
               authMode: authMode, // 传递认证模式
               encryptedPassword: loginData.encryptedPassword || false,
               captcha: loginData.captcha,
               captchaId: loginData.captchaId,
             });

             // 保存Token和用户信息
             set({
               accessToken: response.data.accessToken,
               refreshToken: response.data.refreshToken,
               user: response.data.user,
               isAuthenticated: true,
               authMode: authMode, // 保存认证模式
             });

             // 保存RefreshToken到localStorage
             localStorage.setItem('refreshToken', response.data.refreshToken);

             return response.data;
           } catch (error) {
             console.error('登录失败:', error);
             throw error;
           }
         },

         // ... 其他方法保持不变
       }));
