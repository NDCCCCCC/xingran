       import (

       // TestLoginHandler_Integration 测试登录接口集成
       func TestLoginHandler_Integration(t *testing.T) {
           // 设置Gin为测试模式
           gin.SetMode(gin.TestMode)

           // 创建测试Core
           testCore := setupTestCore(t)

           // 创建测试路由
           router := gin.New()
           SetupAuthRouter(router.Group("/api/v1/system/auth"), testCore)

           t.Run("LocalAuthSuccess", func(t *testing.T) {
               // 创建测试用户
               createTestUser(t, testCore, "testuser", "password123")

               // 构造登录请求
               loginReq := LoginRequest{
                   Username: "testuser",
                   Password: "password123",
                   AuthMode: "local",
               }

               body, _ := json.Marshal(loginReq)
               req, _ := http.NewRequest("POST", "/api/v1/system/auth/login", bytes.NewBuffer(body))
               req.Header.Set("Content-Type", "application/json")

               // 执行请求
               w := httptest.NewRecorder()
               router.ServeHTTP(w, req)

               // 验证响应
               assert.Equal(t, http.StatusOK, w.Code)

               var resp response.Response
               json.Unmarshal(w.Body.Bytes(), &resp)

               assert.Equal(t, float64(0), resp.Code) // 成功

               data := resp.Data.(map[string]interface{})
               assert.NotEmpty(t, data["accessToken"])
               assert.NotEmpty(t, data["refreshToken"])

               user := data["user"].(map[string]interface{})
               assert.Equal(t, "testuser", user["username"])
           })

           t.Run("ADAuthSuccess", func(t *testing.T) {
               // Mock AD认证成功
               setupMockAD(t, testCore, "aduser", "adpassword")

               loginReq := LoginRequest{
                   Username: "aduser",
                   Password: "adpassword",
                   AuthMode: "ad",
               }

               body, _ := json.Marshal(loginReq)
               req, _ := http.NewRequest("POST", "/api/v1/system/auth/login", bytes.NewBuffer(body))
               req.Header.Set("Content-Type", "application/json")

               w := httptest.NewRecorder()
               router.ServeHTTP(w, req)

               assert.Equal(t, http.StatusOK, w.Code)

               var resp response.Response
               json.Unmarshal(w.Body.Bytes(), &resp)

               assert.Equal(t, float64(0), resp.Code)

               // 验证用户已同步到数据库
               var user models.User
               err := testCore.DB.GetDB().Where("username = ?", "aduser").First(&user).Error
               assert.NoError(t, err)
               assert.Equal(t, "ad", user.AuthSource)
           })

           t.Run("HybridAuthFallback", func(t *testing.T) {
               // 不存在的本地用户，但AD用户存在
               setupMockAD(t, testCore, "hybriduser", "hybridpass")

               loginReq := LoginRequest{
                   Username: "hybriduser",
                   Password: "hybridpass",
                   AuthMode: "hybrid",
               }

               body, _ := json.Marshal(loginReq)
               req, _ := http.NewRequest("POST", "/api/v1/system/auth/login", bytes.NewBuffer(body))
               req.Header.Set("Content-Type", "application/json")

               w := httptest.NewRecorder()
               router.ServeHTTP(w, req)

               assert.Equal(t, http.StatusOK, w.Code)
               // 验证降级到AD认证成功
           })

           t.Run("InvalidCredentials", func(t *testing.T) {
               createTestUser(t, testCore, "testuser", "password123")

               loginReq := LoginRequest{
                   Username: "testuser",
                   Password: "wrongpassword",
                   AuthMode: "local",
               }

               body, _ := json.Marshal(loginReq)

               w := httptest.NewRecorder()
               router.ServeHTTP(w, req)

               assert.Equal(t, http.StatusUnauthorized, w.Code)
           })
       }
