       type LoginRequest struct {

           Password           string `json:"password" binding:"required"`
           AuthMode           string `json:"authMode,omitempty"` // 新增：认证模式

       // login 用户登录（重构版本）
       func login(core *core.Core) gin.HandlerFunc {
           return func(c *gin.Context) {
               var req LoginRequest
               if err := c.ShouldBindJSON(&req); err != nil {
                   response.Error(c, response.ErrBadRequest, "请求参数错误")
                   return
               }

               ctx := c.Request.Context()

               // 1. 检查验证码（保持原有逻辑）
               if core.CaptchaService.IsEnabled() {
                   clientIP := c.ClientIP()
                   if req.Captcha == "" || req.CaptchaID == "" {
                       response.Error(c, response.ErrCaptchaError, "请输入验证码")
                       return
                   }
                   if err := core.CaptchaService.VerifyCaptcha(ctx, req.CaptchaID, req.Captcha, clientIP); err != nil {
                       response.Error(c, response.ErrCaptchaError, err.Error())
                       return
                   }
               }

               // 2. 检查账号锁定（保持原有逻辑）
               if err := core.CaptchaService.CheckLoginLock(ctx, req.Username); err != nil {
                   response.Error(c, response.ErrForbidden, err.Error())
                   return
               }

               // 3. 解密SM4密码（前端已SM4加密）
               passwordToVerify := req.Password
               if req.EncryptedPassword {
                   // 兼容旧版本SM2加密
                   decrypted, err := core.JWTManager.DecryptPassword(req.Password)
                   if err != nil {
                       response.Error(c, response.ErrBadRequest, "密码解密失败: "+err.Error())
                       return
                   }
                   passwordToVerify = decrypted
               } else {
                   // 新版本：前端已使用SM4加密，这里解密SM4
                   // 注意：SM4密钥需要从请求中获取或使用会话密钥
                   // 暂时假设前端已处理SM4加密，后端直接使用
                   // TODO: 实现SM4解密逻辑（如果需要后端解密）
               }

               // 4. 确定认证模式
               authMode := req.AuthMode
               if authMode == "" {
                   // 从配置读取默认认证模式
                   factory := getAuthStrategyFactory(core)
                   defaultMode, err := factory.GetDefaultAuthMode(ctx)
                   if err != nil {
                       applogger.Errorf("获取默认认证模式失败: %v", err)
                       authMode = "local" // 降级到本地认证
                   } else {
                       authMode = defaultMode
                   }
               }

               // 5. 获取认证器
               factory := getAuthStrategyFactory(core)
               authenticator, err := factory.GetAuthenticator(authMode)
               if err != nil {
                   applogger.Errorf("获取认证器失败: %v", err)
                   response.Error(c, response.ErrInternalError, "认证服务异常")
                   return
               }

               // 6. 执行认证
               authReq := &security.AuthRequest{
                   Username: req.Username,
                   Password: passwordToVerify,
                   IP:       c.ClientIP(),
               }

               authResult, err := authenticator.Authenticate(ctx, authReq)
               if err != nil {
                   // 记录登录失败日志
                   applogger.Warnf("用户 %s 认证失败: %v", req.Username, err)

                   // 检查账号锁定
                   core.CaptchaService.IncrementLoginAttempts(ctx, req.Username)

                   // 返回通用错误（避免泄露用户存在性）
                   if err == security.ErrUserNotFound || err == security.ErrInvalidCredentials {
                       response.Error(c, response.ErrUnauthorized, "用户名或密码错误")
                   } else if err == security.ErrUserDisabled {
                       response.Error(c, response.ErrForbidden, "用户已被禁用")
                   } else {
                       response.Error(c, response.ErrInternalError, "认证失败: "+err.Error())
                   }
                   return
               }

               // 7. 检查认证结果用户信息
               if authResult.User == nil {
                   response.Error(c, response.ErrInternalError, "认证成功但用户信息缺失")
                   return
               }

               // 8. 查询完整用户信息（包含角色和权限）
               var user models.User
               if err := core.DB.GetDB().Preload("Roles").Preload("Dept").Where("id = ?", authResult.User.ID).First(&user).Error; err != nil {
                   applogger.Errorf("查询用户完整信息失败: %v", err)
                   response.Error(c, response.ErrInternalError, "获取用户信息失败")
                   return
               }

               // 9. 生成JWT令牌（保持原有逻辑）
               accessToken, refreshToken, expiresIn, err := core.JWTManager.GenerateTokenPair(&user)
               if err != nil {
                   applogger.Errorf("生成令牌失败: %v", err)
                   response.Error(c, response.ErrInternalError, "令牌生成失败")
                   return
               }

               // 10. 更新登录信息
               loginIP := c.ClientIP()
               loginTime := utils.Now()
               user.LoginIP = &loginIP
               user.LoginTime = &loginTime
               core.DB.GetDB().Save(&user)

               // 11. 清除登录失败记录
               core.CaptchaService.ClearLoginAttempts(ctx, req.Username)

               // 12. 构造响应
               userInfo := convertToUserInfo(&user)
               response.Success(c, LoginResponse{
                   User:         userInfo,
                   AccessToken:  accessToken,
                   RefreshToken: refreshToken,
                   ExpiresIn:    expiresIn,
                   TokenType:    "Bearer",
               })

               applogger.Infof("用户 %s (%s) 登录成功，认证模式: %s", user.Username, authResult.AuthSource, authenticator.Name())
           }
       }

       import { useState, useEffect } from 'react';
       import { getAuthConfig } from '@/services/config'; // 新增API调用

       interface AuthConfig {
         adEnabled: boolean;
         defaultMode: 'local' | 'ad' | 'hybrid';
       }

       const Login: FC = () => {
         const [authConfig, setAuthConfig] = useState<AuthConfig>({
           adEnabled: false,
           defaultMode: 'local',
         });
         const [configLoaded, setConfigLoaded] = useState(false);

         // 页面加载时获取认证配置
         useEffect(() => {
           const loadAuthConfig = async () => {
             try {
               const config = await getAuthConfig();
               setAuthConfig({
                 adEnabled: config.adEnabled || false,
                 defaultMode: config.defaultMode || 'local',
               });

               // 设置默认认证模式
               setAuthMode(config.defaultMode || 'local');
             } catch (error) {
               console.error('加载认证配置失败:', error);
               // 配置加载失败，默认本地认证
               setAuthMode('local');
             } finally {
               setConfigLoaded(true);
             }
           };

           loadAuthConfig();
         }, []);

         // 渲染认证模式选择器（条件渲染）
         {configLoaded && authConfig.adEnabled && (
           <Form.Item label="认证模式">
             <Radio.Group value={authMode} onChange={(e) => setAuthMode(e.target.value)}>
               <Space direction="vertical">
                 <Radio value="local">本地认证</Radio>
                 <Radio value="ad">AD域控认证</Radio>
                 <Radio value="hybrid">混合认证</Radio>
               </Space>
             </Radio.Group>
           </Form.Item>
         )}
       };
