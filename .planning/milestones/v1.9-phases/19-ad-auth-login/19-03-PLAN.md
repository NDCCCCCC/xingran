       ```sql
       -- Description: 支持AD域控账号登录，添加认证源和AD用户名字段

       -- 添加认证源字段（local/ad）
       ALTER TABLE sys_user

       -- 添加AD用户名字段
       ALTER TABLE sys_user
       ADD COLUMN IF NOT EXISTS ad_username VARCHAR(100);

       -- 添加AD DN字段（可选，用于快速定位AD用户）
       ALTER TABLE sys_user
       ADD COLUMN IF NOT EXISTS ad_dn TEXT;

       -- 创建索引：ad_username（用于快速查找AD用户）

       ON sys_user(ad_username);

       -- 创建索引：auth_source（用于按认证源筛选用户）
       CREATE INDEX IF NOT EXISTS idx_sys_user_auth_source
       ON sys_user(auth_source);

       -- 添加注释
       COMMENT ON COLUMN sys_user.auth_source IS '认证源：local=本地认证, ad=AD域控认证';
       COMMENT ON COLUMN sys_user.ad_username IS 'AD域控用户名';
       COMMENT ON COLUMN sys_user.ad_dn IS 'AD域控DN（Distinguished Name）';

       -- 为现有用户设置默认认证源
       UPDATE sys_user
       SET auth_source = 'local'
       WHERE auth_source IS NULL;

       -- 设置NOT NULL约束
       ALTER TABLE sys_user
       ALTER COLUMN auth_source SET NOT NULL;

       -- 创建唯一约束：同一AD用户只能对应一个本地账号
       CREATE UNIQUE INDEX IF NOT EXISTS idx_sys_user_ad_username_unique
       ON sys_user(ad_username)
       WHERE ad_username IS NOT NULL;
     ```

       ```go
       package system

       import (
           "context"
           "errors"
           "fmt"
           "gorm.io/gorm"
           "github.com/xingran-next/xingran-go-backend/internal/models"
           "github.com/xingran-next/xingran-go-backend/internal/services/addomain"
       )

       // UserSyncService 用户同步服务
       type UserSyncService struct {
               db *gorm.DB
           }

           // NewUserSyncService 创建用户同步服务
           func NewUserSyncService(db *gorm.DB) *UserSyncService {
               return &UserSyncService{db: db}
           }

           // SyncUserFromAD 从AD同步用户到sys_user表
           func (s *UserSyncService) SyncUserFromAD(ctx context.Context, adUser *addomain.ADUser, defaultRoleID string) (*models.User, error) {
               // 使用事务确保原子性
               var user models.User
               err := s.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                   // 1. 检查用户是否已存在（按ad_username查找）
                   err := tx.Where("ad_username = ?", adUser.Username).First(&user).Error

                   if err == nil {
                       // 用户已存在，更新信息
                       return s.updateExistingUser(tx, &user, adUser)
                   }

                   if errors.Is(err, gorm.ErrRecordNotFound) {
                       // 用户不存在，创建新用户
                       return s.createNewUser(tx, adUser, defaultRoleID)
                   }

                   return err
               })

               if err != nil {
                   return nil, fmt.Errorf("同步AD用户失败: %w", err)
               }

               return &user, nil
           }

           // createNewUser 创建新用户
           func (s *UserSyncService) createNewUser(tx *gorm.DB, adUser *addomain.ADUser, defaultRoleID string) error {
               user := &models.User{
                   Username:    adUser.Username,
                   Nickname:    &adUser.DisplayName,
                   Email:       &adUser.Email,
                   Phone:       &adUser.Phone,
                   Status:      models.UserStatusEnabled, // 默认启用
                   AuthSource:  "ad",
                   ADUsername:  &adUser.Username,
                   ADDN:        &adUser.UserDN,
                   InitFlag:    true, // 标记为初始密码，需要修改
                   PwdExpireDays: 90,
               }

               // 设置默认密码（首次登录后要求修改）
               pwdManager := security.NewPasswordManager(nil)
               hashedPassword, err := pwdManager.HashPassword("123456") // 默认密码
               if err != nil {
                   return fmt.Errorf("生成默认密码失败: %w", err)
               }
               user.Password = hashedPassword

               // 设置默认部门（如果配置了）
               if defaultDeptID := s.getDefaultDeptID(); defaultDeptID != "" {
                   user.DeptID = &defaultDeptID
               }

               // 创建用户
               if err := tx.Create(user).Error; err != nil {
                   return fmt.Errorf("创建用户失败: %w", err)
               }

               // 分配默认角色
               if defaultRoleID != "" {
                   if err := s.assignRole(tx, user.ID, defaultRoleID); err != nil {
                           return fmt.Errorf("分配角色失败: %w", err)
                   }
               }

               return nil
           }

           // updateExistingUser 更新已存在的用户
           func (s *UserSyncService) updateExistingUser(tx *gorm.DB, user *models.User, adUser *addomain.ADUser) error {
               // 更新用户信息（保留本地修改的字段如nickname）
               updates := map[string]interface{}{
                   "email":    &adUser.Email,
                   "phone":    &adUser.Phone,
                   "ad_dn":    &adUser.UserDN,
               }

               // 只有当nickname为空时才更新（尊重本地修改）
               if user.Nickname == nil || *user.Nickname == "" {
                   updates["nickname"] = &adUser.DisplayName
               }

               if err := tx.Model(user).Updates(updates).Error; err != nil {
                   return fmt.Errorf("更新用户信息失败: %w", err)
               }

               return nil
           }

           // assignRole 分配角色给用户
           func (s *UserSyncService) assignRole(tx *gorm.DB, userID string, roleID string) error {
               // 使用原生SQL插入角色关联
               sql := `INSERT INTO sys_user_role (user_id, role_id, created_at)
                      VALUES (?, ?, NOW())
                      ON CONFLICT (user_id, role_id) DO NOTHING`

               if err := tx.Exec(sql, userID, roleID).Error; err != nil {
                   return err
               }

               return nil
           }

           // getDefaultDeptID 获取默认部门ID
           func (s *UserSyncService) getDefaultDeptID() string {
               var config models.Config
               err := s.db.Where("config_key = ?", "sys.auth.ad.default_dept_id").First(&config).Error
               if err != nil {
                   return "" // 没有配置默认部门
               }
               return config.ConfigValue
           }
       ```

    关键设计点：
    - 使用数据库事务确保原子性
    - 区分首次登录和再次登录逻辑
    - 首次登录创建用户，设置默认密码（要求修改）
    - 再次登录更新信息（保留本地修改）
    - 支持默认角色和部门配置
