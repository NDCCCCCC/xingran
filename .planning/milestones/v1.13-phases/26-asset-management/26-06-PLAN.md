```sql
-- Migration: Add asset management menu and permissions
-- Purpose: Create menu entries and permissions for asset management module
-- Date: 2026-06-08

-- First, find the operations menu parent_id
-- This script assumes the operations menu exists (menu_name like '运维管理')
-- If not, adjust the parent_id value accordingly

-- Insert asset management menu (一级菜单)
-- Adjust parent_id to match your operations menu ID
INSERT INTO sys_menu (menu_name, parent_id, order_num, path, component, is_frame, is_cache, menu_type, visible, status, perms, icon, created_by, created_at, updated_by, updated_at, remark)
VALUES (
    '资产管理',                   -- menu_name
    (SELECT id FROM sys_menu WHERE menu_name = '运维管理' LIMIT 1), -- parent_id (子菜单)
    5,                           -- order_num (排在工位之后)
    'assets',                    -- path
    'operations/assets/index',   -- component (frontend path)
    0,                           -- is_frame (0=内嵌)
    0,                           -- is_cache (0=不缓存)
    'C',                         -- menu_type (C=菜单)
    '1',                         -- visible (1=显示)
    '0',                         -- status (0=正常)
    '',                          -- perms (菜单本身不需要权限)
    'DatabaseOutlined',          -- icon
    'system',                    -- created_by
    CURRENT_TIMESTAMP,          -- created_at
    '',                          -- updated_by
    CURRENT_TIMESTAMP,           -- updated_at
    '资产管理菜单'                -- remark
)
ON CONFLICT (menu_name, parent_id) DO NOTHING; -- 防止重复插入

-- Get the asset menu id we just created
DO $$
DECLARE
    v_asset_menu_id UUID;
BEGIN
    SELECT id INTO v_asset_menu_id FROM sys_menu WHERE menu_name = '资产管理' AND parent_id IS NOT NULL LIMIT 1;

    -- Insert asset management permissions (按钮权限)

    -- 资产查询
    INSERT INTO sys_menu (menu_name, parent_id, order_num, path, component, is_frame, is_cache, menu_type, visible, status, perms, icon, created_by, created_at, updated_by, updated_at, remark)
    VALUES (
        '资产查询', v_asset_menu_id, 1, '#', '', 1, 0, 'F', '0', '0', 'ops:asset:list', '', 'system', CURRENT_TIMESTAMP, '', CURRENT_TIMESTAMP, ''
    )
    ON CONFLICT (menu_name, parent_id) DO NOTHING;

    -- 资产新增
    INSERT INTO sys_menu (menu_name, parent_id, order_num, path, component, is_frame, is_cache, menu_type, visible, status, perms, icon, created_by, created_at, updated_by, updated_at, remark)
    VALUES (
        '资产新增', v_asset_menu_id, 2, '#', '', 1, 0, 'F', '0', '0', 'ops:asset:add', '', 'system', CURRENT_TIMESTAMP, '', CURRENT_TIMESTAMP, ''
    )
    ON CONFLICT (menu_name, parent_id) DO NOTHING;

    -- 资产修改
    INSERT INTO sys_menu (menu_name, parent_id, order_num, path, component, is_frame, is_cache, menu_type, visible, status, perms, icon, created_by, created_at, updated_by, updated_at, remark)
    VALUES (
        '资产修改', v_asset_menu_id, 3, '#', '', 1, 0, 'F', '0', '0', 'ops:asset:edit', '', 'system', CURRENT_TIMESTAMP, '', CURRENT_TIMESTAMP, ''
    )
    ON CONFLICT (menu_name, parent_id) DO NOTHING;

    -- 资产删除
    INSERT INTO sys_menu (menu_name, parent_id, order_num, path, component, is_frame, is_cache, menu_type, visible, status, perms, icon, created_by, created_at, updated_by, updated_at, remark)
    VALUES (
        '资产删除', v_asset_menu_id, 4, '#', '', 1, 0, 'F', '0', '0', 'ops:asset:delete', '', 'system', CURRENT_TIMESTAMP, '', CURRENT_TIMESTAMP, ''
    )
    ON CONFLICT (menu_name, parent_id) DO NOTHING;

END $$;

-- Add comments for documentation
COMMENT ON TABLE sys_menu IS '菜单权限表：资产管理菜单和按钮权限';
```

Key implementation details:
1. Find operations menu parent_id dynamically (avoids hardcoding)
2. Asset menu as child of operations (运维管理)
3. Four button permissions: list, add, edit, delete
4. Use `ON CONFLICT DO NOTHING` for idempotent migration
5. Icon: DatabaseOutlined (matches asset concept)
6. Component path: `operations/assets/index` matches frontend
7. Menu type C for menu, F for button permissions
8. Order_num: 5 (after workstation, adjust as needed)

Note: If the operations menu doesn't exist, adjust the parent_id query or create the operations menu first.
