```typescript
// sessionStorage hydrate：硬刷新时立即读取缓存数据填充 store
const hydrateFromSessionStorage = () => {
  try {
    const raw = sessionStorage.getItem(MENU_CACHE_KEY);
    if (!raw) return null;
    const cached: CachedMenuData = JSON.parse(raw);
    if (cached.version !== MENU_CACHE_VERSION) return null;
    const age = Date.now() - cached.data.cachedAt;
    if (age > 30 * 60 * 1000) return null; // 30min TTL
    return cached.data;
  } catch { return null; }
};

const cachedData = hydrateFromSessionStorage();
```

2. 在组件顶部（useMemo 之前）添加 hydrate useEffect（替换现有的 fetchAll useEffect）：
```typescript
useEffect(() => {
  if (!cachedData) return;
  // Hydrate store from sessionStorage immediately (no API call)
  useMenuStore.getState().setMenus(cachedData.menus);
  useMenuStore.getState().setPermissions(cachedData.permissions);
}, [cachedData]);
```

3. 修改现有的菜单加载 useEffect（DynamicRoutes:133-140），在缓存存在时跳过 API 调用，缓存不存在时加载 + 回填 sessionStorage：
```typescript
useEffect(() => {
  if (isAuthenticated && initialized) {
    if (cachedData) {
      // Background revalidate（从 sessionStorage 渲染后，后台异步补拉最新数据）
      fetchAll().then((fresh) => {
        if (fresh) {
          sessionStorage.setItem(MENU_CACHE_KEY, JSON.stringify({
            version: MENU_CACHE_VERSION,
            data: {
              menus: useMenuStore.getState().menus,
              allMenus: useMenuStore.getState().allMenus,
              permissions: useMenuStore.getState().permissions,
              cachedAt: Date.now(),
            },
          }));
        }
      }).catch(console.error);
    } else {
      // No cache — load and cache
      fetchAll().then(() => {
        const { menus, allMenus, permissions } = useMenuStore.getState();
        sessionStorage.setItem(MENU_CACHE_KEY, JSON.stringify({
          version: MENU_CACHE_VERSION,
          data: { menus, allMenus, permissions, cachedAt: Date.now() },
        }));
      }).catch(console.error);
    }
  }
}, [isAuthenticated, initialized, fetchAll, cachedData]);
```
