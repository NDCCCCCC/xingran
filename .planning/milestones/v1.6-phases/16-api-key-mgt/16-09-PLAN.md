```go
func setupTestDB(t *testing.T) *gorm.DB {
    // 使用内存数据库或测试数据库
    // 每个测试使用独立事务
    db, _ := gorm.Open(postgres.Open(dsn), &gorm.Config{})
    return db
}

func wrapTestInTransaction(db *gorm.DB, testFunc func(*gorm.DB)) {
    tx := db.Begin()
    defer tx.Rollback()

    testFunc(tx)

    // 回滚事务，不保留测试数据
    tx.Rollback()
}
```
