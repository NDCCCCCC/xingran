  // 根据当前路径查找父菜单的 key
  const findParentKey = (pathname: string): string | null => {
    // 标准化路径函数，统一去掉前导斜杠
    const normalizePath = (p: string): string => {
      return p.startsWith('/') ? p.slice(1) : p;
    };

    const normalizedPathname = normalizePath(pathname);
    console.log('[findParentKey] 查找:', pathname, '标准化后:', normalizedPathname);

    // 调试：打印 menuPathMap 中所有相关的 key
    console.log('[findParentKey] menuPathMap 中的所有 ops 相关 key:', Array.from(menuPathMap.keys()).filter(k => k.includes('ops')));
    console.log('[findParentKey] 是否包含 exact key:', menuPathMap.has('ops/duty/duty/pools'));

    // 使用 menuPathMap 查找当前路径的菜单信息
    const menuInfo = menuPathMap.get(normalizedPathname);
    console.log('[findParentKey] menuPathMap.get结果:', menuInfo);

    if (menuInfo) {
      console.log('[findParentKey] 找到菜单:', normalizedPathname, '层级:', menuInfo.level, '所属一级菜单:', menuInfo.topLevel);

      // 如果是一级菜单，不需要展开父菜单
      if (menuInfo.level === 1) {
        return null;
      }

      // 如果是二级菜单，返回所属的一级菜单 key
      if (menuInfo.level === 2) {
        return menuInfo.topLevel;
      }

      // 如果是三级菜单，返回所属的二级菜单 key
      if (menuInfo.level === 3 && menuInfo.secondLevel) {
        return menuInfo.secondLevel;
      }
    }

    console.log('[findParentKey] 未找到菜单:', normalizedPathname);
    return null;
  };
